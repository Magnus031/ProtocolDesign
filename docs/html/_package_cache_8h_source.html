20 inline bool _Download_Package(const ConstStringS& cstrPackageRoot, const ConstStringS& cstrSrcFileName,
21  bool bForce, const _NetCfg& cfg, UniqueStringS& strDlFile)
22 {
23  //check download
24  _SourceHelper::GenerateDownloadFileName(cstrPackageRoot, cstrSrcFileName, strDlFile); //may throw
25  FsPathHelper::ToPlatform(strDlFile.GetAddress());
26  if ( !bForce && FsOpHelper::CheckFileExists(strDlFile.GetAddress()) )
27  return true;
28  //url
29  uintptr uCount = cfg.GetCount();
30  if ( uCount == 0 ) {
31  ConsoleHelper::WriteLine(_S("ERROR: there is no network configuration!"));
32  return false;
33  }
34  UniqueStringS strUrl;
35  for ( uintptr i = 0; i < uCount; ++ i ) {
36  strUrl = U2S(UniqueStringHelper::To_ConstString(cfg.GetUrl(i))).V(); //may throw
37  uintptr uLength = strUrl.GetLength();
38  assert( uLength != 0 );
39  if ( *(strUrl.GetAddress() + uLength - 1) != _S('/') )
40  strUrl.Append(1, _S('/')); //may throw
41  strUrl.Append(cstrSrcFileName.GetLength(), cstrSrcFileName.GetAddress()); //may throw
42  if ( WebHelper::FetchFile(UniqueStringHelper::To_ConstString(strUrl), UniqueStringHelper::To_ConstString(strDlFile)) )
43  return true;
44  }
45  //error
46  ConsoleHelper::Write(_S("ERROR: cannot download ["));
47  ConsoleHelper::Write(cstrSrcFileName.GetAddress());
48  ConsoleHelper::WriteLine(_S("]!"));
49  return false;
50 }

87 inline bool _Load_Packages_To_Cache(const ConstStringS& cstrPackageRoot, UniqueStringA&& strSig, bool bForce,
88  const _NetCfg& cfg, _DependencyGraph& graph)
89 {
90  //task
91  UniqueArray<_DependencyGraph::VertexPosition> arrTask;
92 
93  {
94  _PackageSignature ks;
95  if ( !ks.Set(rv_forward(strSig)) ) { //may throw
96  ConsoleHelper::Write(_S("ERROR: invalid package signature: ["));
97  ConsoleHelper::Write(U2S(rv_forward(strSig)).C().GetAddress()); //may throw
98  ConsoleHelper::WriteLine(_S("]!"));
99  return false;
100  }
101  auto pos = graph.FindVertex(ks.GetSignature());
102  if ( pos.IsNull() ) {
103  pos = graph.AddVertex(rv_forward(ks.GetSignature())); //may throw
104  graph.GetVertexName(pos) = rv_forward(ks.GetName());
105  graph.GetVertexType(pos) = rv_forward(ks.GetType());
106  arrTask.Add(pos); //may throw
107  }
108  } //end block
109 
110  UniqueStringS strCacheDir;
111  UniqueStringS strCacheFile;
112  UniqueStringS strDownDir;
113  UniqueStringS strDownFile;
114  UniqueStringS strName;
115  UniqueStringS strType;
116  UniqueStringS strSrcFileName;
117 
118  _PackageInfo info;
119 
120  //loop
121  while ( !arrTask.IsEmpty() ) {
122  auto pos = arrTask[0];
123  arrTask.Delete(0);
124  strName = U2S(UniqueStringHelper::To_ConstString(graph.GetVertexName(pos))).V(); //may throw
125  strType = U2S(UniqueStringHelper::To_ConstString(graph.GetVertexType(pos))).V(); //may throw
126  _FileNameHelper::GeneratePackageFileName(UniqueStringHelper::To_ConstString(strName),
127  UniqueStringHelper::To_ConstString(strType), strSrcFileName); //may throw
128  //source file in cache
129  _SourceHelper::GenerateCacheFileName(cstrPackageRoot,
130  UniqueStringHelper::To_ConstString(strSrcFileName),
131  strCacheFile); //may throw
132  FsPathHelper::ToPlatform(strCacheFile.GetAddress());
133  if ( !FsOpHelper::CheckFileExists(strCacheFile.GetAddress()) ) {
134  if ( !_Download_Package(cstrPackageRoot, UniqueStringHelper::To_ConstString(strSrcFileName), bForce, cfg, strDownFile) ) //may throw
135  return false;
136  //move
137  if ( !FsOpHelper::Rename(strDownFile.GetAddress(), strCacheFile.GetAddress()) ) {
138  ConsoleHelper::Write(_S("ERROR: cannot move file ["));
139  ConsoleHelper::Write(strSrcFileName.GetAddress());
140  ConsoleHelper::WriteLine(_S("]!"));
141  return false;
142  }
143  //info
144  if ( !_Load_Info_Expand_Dir(cstrPackageRoot,
145  UniqueStringHelper::To_ConstString(strName),
146  UniqueStringHelper::To_ConstString(strType),
147  DECLARE_TEMP_CACHE_NAME, info, strCacheDir) ) //may throw
148  return false;
149  }
150  else {
151  if ( bForce ) {
152  if ( !_Download_Package(cstrPackageRoot, UniqueStringHelper::To_ConstString(strSrcFileName), bForce, cfg, strDownFile) ) //may throw
153  return false;
154  //info
155  if ( !_Load_Info_Expand_Dir(cstrPackageRoot,
156  UniqueStringHelper::To_ConstString(strName),
157  UniqueStringHelper::To_ConstString(strType),
158  DECLARE_TEMP_CACHE_NAME, info, strCacheDir) ) //may throw
159  return false;
160  _PackageInfo infoD;
161  if ( !_Load_Info_Expand_Dir(cstrPackageRoot,
162  UniqueStringHelper::To_ConstString(strName),
163  UniqueStringHelper::To_ConstString(strType),
164  DECLARE_TEMP_DOWNLOAD_NAME, infoD, strDownDir) ) //may throw
165  return false;
166  //delete
167  if ( !DirectoryHelper::DeleteAll(UniqueStringHelper::To_ConstString(strDownDir), true) ) {
168  ConsoleHelper::Write(_S("ERROR: cannot delete directory ["));
169  ConsoleHelper::Write(strDownDir.GetAddress());
170  ConsoleHelper::WriteLine(_S("]!"));
171  return false;
172  }
173  //compare
174  if ( info.CompareVersion(infoD) < 0 ) {
175  //move
176  if ( !_Delete_File(UniqueStringHelper::To_ConstString(strCacheFile)) ) {
177  ConsoleHelper::Write(_S("ERROR: cannot delete file ["));
178  ConsoleHelper::Write(strCacheFile.GetAddress());
179  ConsoleHelper::WriteLine(_S("]!"));
180  return false;
181  }
182  if ( !FsOpHelper::Rename(strDownFile.GetAddress(), strCacheFile.GetAddress()) ) {
183  ConsoleHelper::Write(_S("ERROR: cannot move file ["));
184  ConsoleHelper::Write(strSrcFileName.GetAddress());
185  ConsoleHelper::WriteLine(_S("]!"));
186  return false;
187  }
188  info = rv_forward(infoD);
189  }
190  else {
191  //delete
192  if ( !_Delete_File(UniqueStringHelper::To_ConstString(strDownFile)) ) {
193  ConsoleHelper::Write(_S("ERROR: cannot delete file ["));
194  ConsoleHelper::Write(strDownFile.GetAddress());
195  ConsoleHelper::WriteLine(_S("]!"));
196  return false;
197  }
198  }
199  }
200  else {
201  //info
202  if ( !_Load_Info_Expand_Dir(cstrPackageRoot,
203  UniqueStringHelper::To_ConstString(strName),
204  UniqueStringHelper::To_ConstString(strType),
205  DECLARE_TEMP_CACHE_NAME, info, strCacheDir) ) //may throw
206  return false;
207  }
208  }
209  //installed
210  _InstalledPackageItem item;
211  if ( !item.Fetch(cstrPackageRoot,
212  UniqueStringHelper::To_ConstString(strName),
213  UniqueStringHelper::To_ConstString(strType)) ) { //may throw
214  graph.SetVertexAction(pos, PACT_NONE);
215  }
216  else {
217  if ( item.GetPackageInfo().CompareVersion(info) < 0 )
218  graph.SetVertexAction(pos, PACT_UPGRADE);
219  else
220  graph.SetVertexAction(pos, PACT_INSTALLED);
221  }
222  //dependencies
223  for ( uintptr i = 0; i < info.GetDependencyCount(); ++ i ) {
224  _PackageSignature& ks(info.GetDependencyAt(i));
225  auto pos1 = graph.FindVertex(ks.GetSignature());
226  if ( pos1.IsNull() ) {
227  pos1 = graph.AddVertex(rv_forward(ks.GetSignature())); //may throw
228  graph.GetVertexName(pos1) = rv_forward(ks.GetName());
229  graph.GetVertexType(pos1) = rv_forward(ks.GetType());
230  }
231  graph.AddEdge(pos, pos1); //may throw
232  arrTask.Add(pos1); //may throw
233  }
234  } //end while
235 
236  return true;
237 }
