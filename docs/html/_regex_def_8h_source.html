44  {
45  //init
46  if( m_arr.IsBlockNull() )
47  m_arr = GKC::ShareArrayHelper::MakeShareArray<_RegexCharRange>(GKC::MemoryHelper::GetCrtMemoryManager()); //may throw
48  //add
49  if( m_arr.GetCount() == 0 ) {
50  m_arr.Add(rcr); //may throw
51  return ;
52  }
53  //check ranges
54  uintptr index = 0;
55  while( index < m_arr.GetCount() ) {
56  _RegexCharRange& rcrC = m_arr[index];
57  //left
58  if( rcr.uHigh < rcrC.uLow - 1 )
59  break;
60  //overlapped
61  if( rcr.uHigh <= rcrC.uHigh ) {
62  //combine
63  if( rcrC.uLow > rcr.uLow )
64  rcrC.uLow = rcr.uLow;
65  return ;
66  }
67  //extended
68  if( rcr.uLow <= rcrC.uHigh + 1 ) {
69  //combine
70  if( rcrC.uLow > rcr.uLow )
71  rcrC.uLow = rcr.uLow;
72  rcrC.uHigh = rcr.uHigh;
73  //next
74  uintptr indexNext = index + 1;
75  while( indexNext < m_arr.GetCount() ) {
76  _RegexCharRange& rcrCurrent = m_arr[index];
77  _RegexCharRange& rcrNext = m_arr[indexNext];
78  if( rcrCurrent.uHigh < rcrNext.uLow - 1 )
79  break;
80  //combine
81  if( rcrCurrent.uHigh < rcrNext.uHigh )
82  rcrCurrent.uHigh = rcrNext.uHigh;
83  m_arr.RemoveAt(indexNext);
84  }
85  return ;
86  }
87  //right
88  index ++;
89  } //end while
90  //insert
91  m_arr.InsertAt(index, 1, rcr); //may throw

150  void CalcComplement(_RegexCharRangeSet& rs) const
151  {
152  if( rs.m_arr.IsBlockNull() )
153  rs.m_arr = GKC::ShareArrayHelper::MakeShareArray<_RegexCharRange>(GKC::MemoryHelper::GetCrtMemoryManager()); //may throw
154  //clear
155  rs.m_arr.RemoveAll();
156  //empty
157  const uint c_Max = 0xFFFFFFFE;
158  uint uLeft = 0;
159  _RegexCharRange rcrNow;
160  uintptr uCount = GetCount();
161  if( uCount == 0 ) {
162  rcrNow.uLow = uLeft + 1;
163  rcrNow.uHigh = c_Max;
164  rs.m_arr.Add(rcrNow); //may throw
165  return ;
166  }
167  //loop
168  for( uintptr i = 0; i < uCount; i ++ ) {
169  const _RegexCharRange& rcr = GetRange(i);
170  if( rcr.uLow <= 1 ) {
171  uLeft = rcr.uHigh;
172  continue;
173  }
174  rcrNow.uLow = uLeft + 1;
175  rcrNow.uHigh = rcr.uLow - 1;
176  rs.m_arr.Add(rcrNow); //may throw
177  uLeft = rcr.uHigh;
178  }
179  //last
180  if( uLeft < c_Max ) {
181  rcrNow.uLow = uLeft + 1;
182  rcrNow.uHigh = c_Max;
183  rs.m_arr.Add(rcrNow); //may throw
184  }
185  }

354 inline GKC::AstTree::Iterator _RegexCharRangeSet_To_AST(const _RegexCharRangeSet& rs, GKC::AstTree& tree)
355 {
356  uintptr uCount = rs.GetCount();
357  assert( uCount > 0 );
358  if( uCount == 1 ) {
359  //only a leaf
360  const _RegexCharRange& rcr = rs.GetRange(0);
361  return _RegexCharRange_To_AST(rcr, tree); //may throw
362  }
363  //OR tree
364  GKC::AstTree::Iterator iterOP(tree.CreateNode(0, REGEX_OP_OR)); //may throw
365  GKC::AstTree::Iterator iterLast(tree.GetNullIterator());
366  for( uintptr i = 0; i < uCount; i ++ ) {
367  const _RegexCharRange& rcr = rs.GetRange(i);
368  auto iterLeaf(_RegexCharRange_To_AST(rcr, tree)); //may throw
369  tree.SetParent(iterLeaf, iterOP);
370  if( iterLast.IsNull() )
371  tree.SetChild(iterOP, iterLeaf);
372  else
373  tree.SetNext(iterLast, iterLeaf);
374  iterLast = iterLeaf;
375  } //end for
376  return iterOP;
377 }
