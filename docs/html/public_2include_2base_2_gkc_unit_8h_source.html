220  {
221  //const strings
222  DECLARE_LOCAL_CONST_STRING(CharS, l_szSep1, l_uSep1Len, _S("=========================="))
223  DECLARE_LOCAL_CONST_STRING(CharS, l_szSep2, l_uSep2Len, _S("**************************"))
224  DECLARE_LOCAL_CONST_STRING(CharS, l_szSep3, l_uSep3Len, _S("--------------------------"))
225  DECLARE_LOCAL_CONST_STRING(CharS, l_szColon, l_uColonLen, _S(":"))
226  DECLARE_LOCAL_CONST_STRING(CharS, l_szNoTest, l_uNoTestLen, _S("ERROR: NO SUCH TEST!"))
227 
228  (void)l_uSep1Len;
229  (void)l_uSep2Len;
230  (void)l_uSep3Len;
231  (void)l_uColonLen;
232  (void)l_uNoTestLen;
233 
234  //map
235  _UnitTestMap& map = _UnitTestMapHelper::GetUnitTestMap();
236  uintptr uTotal = 0;
237  uintptr uFailed = 0;
238 
239  if( args.GetCount() <= 1 ) {
240  //all tests
241  _UnitTestMap::ItemInfo info;
242  bool bContinue = map.EnumFirst(info);
243  while( bContinue ) {
244  //current test
245  ConsoleHelper::WriteLine(l_szSep1);
246  ConsoleHelper::Write(info.szName);
247  ConsoleHelper::WriteLine(l_szColon);
248  if( !info.pFunc(buffer) ) {
249  ConsoleHelper::WriteLine(buffer.GetAddress());
250  uFailed ++;
251  }
252  ConsoleHelper::WriteLine(l_szSep2);
253  uTotal ++;
254  bContinue = map.EnumNext(info);
255  //sleep
256  ThreadHelper::Sleep(1);
257  }
258  }
259  else {
260  UniqueStringS strM;
261  //specified tests
262  auto iter(args.GetBegin());
263  assert( iter != args.GetEnd() );
264  iter.MoveNext(); //from 1
265  for( ; iter != args.GetEnd(); iter.MoveNext() ) {
266  //current test
267  ConsoleHelper::WriteLine(l_szSep1);
268  ConsoleHelper::Write(iter.get_Value().GetAddress());
269  ConsoleHelper::WriteLine(l_szColon);
270  //find
271  strM.DeleteAll();
272  strM.Append(iter.get_Value().GetLength(), iter.get_Value().GetAddress()); //may throw
273  _UnitTestFunc pFunc = map.Find(strM);
274  if( pFunc == NULL ) {
275  ConsoleHelper::WriteLine(l_szNoTest);
276  uFailed ++;
277  }
278  else {
279  //run test
280  if( !pFunc(buffer) ) {
281  ConsoleHelper::WriteLine(buffer.GetAddress());
282  uFailed ++;
283  }
284  }
285  ConsoleHelper::WriteLine(l_szSep2);
286  uTotal ++;
287  }
288  } //end if
289 
290  //total information
291  {
292  int ret = value_to_string(buffer.GetAddress(), _UnitTestMessageBuffer::c_size,
293  _S("Total (%zu), Failed (%zu)"), uTotal, uFailed);
294  buffer.SetLength(VTOS_LENGTH_FROM_RETURN(ret));
295  ConsoleHelper::WriteLine(l_szSep3);
296  ConsoleHelper::WriteLine(buffer.GetAddress());
297  } //end block
298 
299  return uFailed != 0 ? 1 : 0;
300  }
301 };
