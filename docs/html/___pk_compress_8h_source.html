34 /*
35 if matching length > 9, output 010, output 1 byte to represent actual length.
36 0 -- 10, 1 -- 11, ..., 252 -- 262. (106H)
37 253, 254, 255 are used for special flags.
38 */
39 #define LEN_LAST_ELEMENT (9 - 1) // (numofarray(g_EncodeLenBitsTable) - 1)
40 #define VL_THRESHOLD (9 + 1) // (numofarray(g_EncodeLenBitsTable) + 1)
41 
42 #define MAX_MATCH_LEN (252 + VL_THRESHOLD)
43 
44 /*
45 The maximum value of high-order position byte is 1FH,
46 then the maximum position is 1FFFH, i.e., 2000H - 1, MAX_MATCH_RANGE - 1
47 */

49 // pkzip_buffer
50 
51 /*
52 working buffer layout:
53 ----------
54 RawBuf --- processed data, size=0x2200, the maximum matching range 0x2000 + step transition area 0x200
55 CodingBuf --- encoding data, at least 0x2000, the maximum matching range
56 Extra --- extension area for reading input file
57 ----------
58 CodedBuf --- encoded data, >=0x400
59 
60 RawBufSize + CodingBufSize >=0x4200 for zip search table
61 CodingBufSize : CodingBuf + Extra
62 
63 After encoded, data of CodingBuf is moved to RawBuf, data of extra area is moved to CodingBuf
64 */
65 
66 struct pkzip_buffer
67 {
68  enum { EXTRA_SIZE = 0x9600 };
69  //0xD800
70  byte data_buffer[MAX_MATCH_RANGE + MATCH_CONNECT_SIZE //RawBuf
71  + MAX_MATCH_RANGE //CodingBuf
72  + EXTRA_SIZE]; //ExtraBuf
73  byte* pRawBuf = data_buffer;
74  byte* pCodingBuf = &data_buffer[MAX_MATCH_RANGE + MATCH_CONNECT_SIZE]; //0x2200
75  ushort uCodingBufSize = MAX_MATCH_RANGE + EXTRA_SIZE;
76 
77  byte code_buffer[0xD800];
78  byte* pCodedBuf = code_buffer;
79  ushort uCodedBufSize = 0xD800 - 0x10;
80 
81  ushort uCodeStoreOfs; //next position for storing data (encode: in CodedBuf, decode: in RawBuf)
82  ushort uBytesReadInBuf; //the number of bytes of input data (encode: in CodingBuf, decode: in CodedBuf)
83  ushort uNextSrcDataOfs; //the offset of next source data (encode: in CodingBuf, decode: in CodedBuf)
84 };

88 struct pkzip_putbuf
89 {
90  //flag & data output buffer
91  //2: ushort, flag buffer
92  //16: flag buffer has 16 bits, data buffer must have 16 bytes for non-encoded characters.
93  // it may be 1 byte if length > 9 and 1 low byte of position for encoded characters.
94  byte put_buffer[2 + 16 * 2];
95  byte uPutLen; //left bits in Flag Buffer (ushort), 0---16
96  ushort uPutBufIdx; //index of output data buffer
97  byte* pPutBuf = put_buffer;
98  ushort* pFlagBuf = (ushort*)put_buffer;
99 };

103 class pkzip_compressor
104 {
105 private:
106  //hash bucket (pointer group array)
107  // byte offsets for search table (5+4)*0x100 = 256*9
108  enum { ZIP_SYMB_GROUPS = 0x900 };
109  ushort m_ZipGroupOfsTable[ZIP_SYMB_GROUPS];
110  // offsets for input buffer 0x4200
111  ushort m_ZipSearchTable[2 * MAX_MATCH_RANGE + MATCH_CONNECT_SIZE];
112 
113  //buffers
114  pkzip_buffer m_Buf;
115  pkzip_putbuf m_Put;
116 
117  ushort m_uCodingBufIdx; //the index of position for next input data moved to CodingBuf [0 -- MAX_MATCH_RANGE]
118  ushort m_uNextEncodeOfs; //the offset of next encoding string base on RawBuf
119  ushort m_uEncodeMatchEndOfs;
120  ushort m_uSearchEndOfs;
121 
122  bool m_bDataMoved;

134 private:
135  ushort& group_ofs_to_buffer_ofs(ushort uGroupOfs) noexcept
136  {
137  return *((ushort*)((byte*)m_ZipSearchTable + uGroupOfs));
138  }
139 
140  //(uSrcEndOfs - uSrcOfs) must be <= 65535 / 2
141  //uSrcOfs: the start offset of input data (RawBuf)
142  //uSrcEndOfs: the end offset of input data, the character pointed to is valid (RawBuf)
143  void build_zip_search_table(ushort uSrcOfs, ushort uSrcEndOfs) noexcept
144  {
145  ushort size = uSrcEndOfs - uSrcOfs; //size: buffer_size-1, must be >= 1
146  mem_zero(m_ZipGroupOfsTable, sizeof(m_ZipGroupOfsTable));
147  ushort i, j;
148  byte x, y;
149  const byte* pSrc;
150  //calculate bytes for each bucket element
151  pSrc = m_Buf.pRawBuf + uSrcOfs;
152  j = size;
153  x = *pSrc ++;
154  do {
155  y = *pSrc ++;
156  m_ZipGroupOfsTable[((ushort)x + (ushort)y) * 4 + (ushort)x] += 2; //hash value: 5*x+4*y
157  x = y;
158  } while( --j );
159  //each bucket element points to the end of region of search table
160  i = 0;
161  j = 0;
162  do {
163  m_ZipGroupOfsTable[j] = (i += m_ZipGroupOfsTable[j]);
164  } while( ++j < ZIP_SYMB_GROUPS ); // 900H
165  //fill reversely
166  //each bucket element points to the start of region of search table
167  //offsets are in incremental order in region
168  pSrc = m_Buf.pRawBuf + uSrcEndOfs;
169  i = uSrcEndOfs;
170  j = size;
171  y = *pSrc --;
172  do {
173  i --;
174  x = *pSrc --;
175  group_ofs_to_buffer_ofs(m_ZipGroupOfsTable[(((ushort)x + (ushort)y) << 2) + (ushort)x] -= 2) = i;
176  y = x;
177  } while( --j );
178  }
179 
180  //only encode string if the matching length >= 2
181  ushort zip_search(ushort uSrcOfs, ushort uSrcEndOfs, ushort& uMatchRelaOfs) noexcept
182  {
183  ushort size = uSrcEndOfs - uSrcOfs; // >=1
184  byte x, y;
185  x = *(m_Buf.pRawBuf + uSrcOfs);
186  y = *(m_Buf.pRawBuf + uSrcOfs + 1);
187  ushort uGroupOfsTblIdx = (((ushort)x + (ushort)y) << 2) + (ushort)x;
188  ushort uGroupOfs = m_ZipGroupOfsTable[uGroupOfsTblIdx];
189  //no match after uSrcOfs
190  if( group_ofs_to_buffer_ofs(uGroupOfs) >= uSrcOfs ) return 0;
191  //current maximum matching length, must be > 2
192  ushort uMatchLen = 2; //match the last two characters, assume matching at least two characters
193  uMatchRelaOfs = 0; //initialize
194  //the beginning offset for searching, uSrcOfs - uMatchBeginOfs < MAX_MATCH_RANGE
195  // before uSrcOfs
196  ushort uMatchBeginOfs = uSrcOfs - (MAX_MATCH_RANGE - 1);
197  //check searching range from bucket start, revise group offset table
198  if( group_ofs_to_buffer_ofs(uGroupOfs) < uMatchBeginOfs ) {
199  do {
200  uGroupOfs += 2; //ushort*
201  } while( group_ofs_to_buffer_ofs(uGroupOfs) < uMatchBeginOfs );
202  m_ZipGroupOfsTable[uGroupOfsTblIdx] = uGroupOfs; //change first position
203  }
204  //search
205  for( ; ; uGroupOfs += 2 ) { //each offset in searching table
206  ushort uCmpDataOfs = group_ofs_to_buffer_ofs(uGroupOfs);
207  if( uSrcOfs <= uCmpDataOfs ) { //after
208  if( uMatchRelaOfs != 0 ) return uMatchLen; //return matching length if offset is set
209  else return 0; //no match
210  }
211  //compare the last two characters in matched string
212  if( *((ushort*)(m_Buf.pRawBuf + uSrcOfs + uMatchLen - 2)) !=
213  *((ushort*)(m_Buf.pRawBuf + uCmpDataOfs + uMatchLen - 2)) )
214  continue;
215  //compare the first character
216  if( x != *(m_Buf.pRawBuf + uCmpDataOfs) )
217  continue;
218  //next characters from 2
219  ushort i = 2;
220  while( 1 ) {
221  //maximum search length
222  if( i >= (size + 1) )
223  break;
224  //compare
225  if( *(m_Buf.pRawBuf + uSrcOfs + i) !=
226  *(m_Buf.pRawBuf + uCmpDataOfs + i) )
227  break;
228  //maximum matching length
229  if( ++i == MAX_MATCH_LEN ) {
230  uMatchRelaOfs = uSrcOfs - uCmpDataOfs;
231  return MAX_MATCH_LEN;
232  }
233  }
234  //update if the length of this time is maximum
235  //this time is also the nearest matched string to the uSrcOfs
236  if( uMatchLen > i ) continue;
237  uMatchLen = i;
238  uMatchRelaOfs = uSrcOfs - uCmpDataOfs;
239  }
240  }

242  //bits: output bit string, length or high byte of position, bit 0->15
243  //len: the number of bits to write
244  void put_bit(ushort bits, ushort len) noexcept
245  {
246  if( m_Put.uPutLen != 0 ) { //rest bits
247  *(m_Put.pFlagBuf) |= (bits << (16 - m_Put.uPutLen)); //left shift
248  }
249  if( m_Put.uPutLen > len ) {
250  m_Put.uPutLen -= len; //enough room
251  }
252  else {
253  if( !byte_order_helper::IsBigEndianHost() ) *(m_Put.pFlagBuf) = byte_order_helper::Swap(*(m_Put.pFlagBuf));
254  mem_copy(m_Put.pPutBuf, m_Put.uPutBufIdx, m_Buf.pCodedBuf + m_Buf.uCodeStoreOfs); //flag & data buffer are moved to CodedBuf
255  m_Buf.uCodeStoreOfs += m_Put.uPutBufIdx;
256  m_Put.uPutBufIdx = 2; //reset position of output data buffer
257  *(m_Put.pFlagBuf) = bits >> m_Put.uPutLen; //rest bit string
258  m_Put.uPutLen += (16 - len);
259  }
260  }
261 
262  //if len == 1, data is non-encoded byte.
263  //if len > 1, data is position, len is length.
264  void encode(ushort data, ushort len) noexcept
265  {
266  if( len == 1 ) { // output the byte stored in `data`
267  put_bit(0, 1); //write a bit `0` to FlagBuf
268  }
269  else {
270  put_bit(1, 1);
271  if( VL_THRESHOLD > len ) { //length range 2--9, using length encoded table
272  put_bit(g_EncodeLenBitsTable[len - 2], g_EncodeLenLenTable[len - 2]);
273  }
274  else { //length > 9
275  //Length > the number of items in encoding tables, variable length
276  put_bit(g_EncodeLenBitsTable[LEN_LAST_ELEMENT], g_EncodeLenLenTable[LEN_LAST_ELEMENT]);
277  m_Put.pPutBuf[m_Put.uPutBufIdx ++] = len - VL_THRESHOLD;
278  }
279  if( len != 2 ) { //high-order position byte, non-encoding for len == 2 (postion < 100H)
280  //encode position
281  put_bit(g_EncodePosBitsTable[data >> 8], g_EncodePosLenTable[data >> 8]);
282  }
283  }
284  m_Put.pPutBuf[m_Put.uPutBufIdx ++] = (byte)data; //low-order position byte
285  }
286 
287  //output end flag
288  void encode_end() noexcept
289  {
290  put_bit(1, 1);
291  put_bit(g_EncodeLenBitsTable[LEN_LAST_ELEMENT], g_EncodeLenLenTable[LEN_LAST_ELEMENT]); //length > 9
292  m_Put.pPutBuf[m_Put.uPutBufIdx ++] = 0xFF; //end flag
293  //flush
294  if( !byte_order_helper::IsBigEndianHost() ) *(m_Put.pFlagBuf) = byte_order_helper::Swap(*(m_Put.pFlagBuf));
295  mem_copy(m_Put.pPutBuf, m_Put.uPutBufIdx, m_Buf.pCodedBuf + m_Buf.uCodeStoreOfs);
296  m_Buf.uCodeStoreOfs += m_Put.uPutBufIdx;
297  }
298 
299  //call this method after calling encode(...)
300  bool is_output_full() const noexcept
301  {
302  //after calling encode(...), the number of pending bytes in tail is (1 * 0x22)
303  //after calling encode_end(...), PutBuf may be flushed two times. So the number of pending bytes in tail is (2 * 0x22)
304  // 2+16*2=0x22
305  return m_Buf.uCodeStoreOfs > (m_Buf.uCodedBufSize - 2 * 0x22); // m_Buf.uCodedBufSize - 0x32
306  }
307 
308  bool need_input() const noexcept
309  {
310  return m_uCodingBufIdx < MAX_MATCH_RANGE;
311  }
312  //move data from extra buffer to CodingBuf
313  void source_fill_coding() noexcept
314  {
315  m_uCodingBufIdx = 0;
316  if( m_Buf.uBytesReadInBuf == 0 ) { //rest bytes in extra buffer
317  return ;
318  }
319  ushort uBytesToMove;
320  uBytesToMove = numeric_min<ushort>(MAX_MATCH_RANGE - m_uCodingBufIdx, m_Buf.uBytesReadInBuf);
321  mem_copy(m_Buf.pCodingBuf + m_Buf.uNextSrcDataOfs, uBytesToMove, m_Buf.pCodingBuf + m_uCodingBufIdx);
322  m_Buf.uBytesReadInBuf -= uBytesToMove; //adjust number of bytes in extra buffer
323  m_Buf.uNextSrcDataOfs += uBytesToMove; //adjust source data offset in extra buffer
324  m_uCodingBufIdx += uBytesToMove;
325  }

350  //return : 0 -- end, 1 -- need input, 2 -- output is full
351  int zip_encode() noexcept
352  {
353  if( m_uEndOfSrcFill == 0 ) {
354  if( need_input() ) return 1;
355  }
356  if( is_output_full() ) return 2;
357 
358  //loop
359  while( 1 ) {
360  //not in encoding loop
361  if( m_uNextEncodeOfs >= m_uEncodeMatchEndOfs ) {
362  if( m_uEndOfSrcFill == 0 && !m_bDataMoved ) {
363  //move processed data
364  if( m_uEncodeStep != 0 ) {
365  mem_move(m_Buf.pRawBuf + MAX_MATCH_RANGE, MAX_MATCH_RANGE + MATCH_CONNECT_SIZE, m_Buf.pRawBuf); //move data from CodingBuf to RawBuf
366  m_uNextEncodeOfs -= MAX_MATCH_RANGE;
367  m_uEncodeMatchEndOfs = m_uNextEncodeOfs;
368  m_uCodingBufIdx = 0;
369  }
370  //fill data
371  if( m_uCodingBufIdx < MAX_MATCH_RANGE ) {
372  source_fill_coding();
373  m_bDataMoved = true;
374  if( need_input() ) return 1;
375  }
376  }
377 
378  m_bDataMoved = false;
379  m_uEncodeMatchEndOfs = MAX_MATCH_RANGE + MATCH_CONNECT_SIZE + m_uCodingBufIdx;
380 
381  //step 4
382  ushort size = m_uEncodeMatchEndOfs - m_uNextEncodeOfs;
383  if( m_uEndOfSrcFill != 0 && size < 2 ) { // 0 or 1
384  if( size == 1 ) {
385  encode(*(m_Buf.pRawBuf + m_uNextEncodeOfs), 1); //non-encoding
386  m_uNextEncodeOfs ++;
387  if( is_output_full() ) return 2;
388  }
389  //output end flag
390  encode_end();
391  return 0;
392  }
393 
394  m_uSearchEndOfs = m_uEncodeMatchEndOfs - 1;
395  if( m_uEndOfSrcFill == 0 ) m_uEncodeMatchEndOfs -= MAX_MATCH_LEN; //not step 4
396 
397  //steps
398 /*
399 Maximum length of search table: RawBuf + CodingBuf, 0x4200
400 0: beginning of CodingBuf -> m_uEncodeMatchEndOfs - 1
401 1: last non-encoding position - 0x2000 + MAX_MATCH_LEN -> m_uEncodeMatchEndOfs - 1, after data moved to RawBuf
402 2: last non-encoding position - 0x2000 -> m_uEncodeMatchEndOfs - 1, after data moved to RawBuf
403 3: start position of any previous step -> position of data end in CodingBuf, when m_uEndOfSrcFill == 1
404 */
405  if( m_uEncodeStep == 2 ) {
406  build_zip_search_table(m_uNextEncodeOfs - MAX_MATCH_RANGE, m_uEncodeMatchEndOfs - 1);
407  }
408  else if( m_uEncodeStep == 1 ) {
409  build_zip_search_table(m_uNextEncodeOfs - MAX_MATCH_RANGE + MAX_MATCH_LEN, m_uEncodeMatchEndOfs - 1);
410  m_uEncodeStep ++;
411  }
412  else { //==0
413  build_zip_search_table(m_uNextEncodeOfs, m_uEncodeMatchEndOfs - 1);
414  m_uEncodeStep ++;
415  }
416  }
417 
418  //compressing loop
419  do {
420  //step 4, last byte
421  if( m_uEndOfSrcFill != 0 && m_uEncodeMatchEndOfs - m_uNextEncodeOfs < 2 ) {
422  assert( m_uEncodeMatchEndOfs - m_uNextEncodeOfs == 1 );
423  encode(*(m_Buf.pRawBuf + m_uNextEncodeOfs), 1); //non-encoding
424  m_uNextEncodeOfs ++;
425  if( is_output_full() ) return 2;
426  continue;
427  }
428 
429  if( m_bNeedSearch ) {
430  if( (m_uMatchLen = zip_search(m_uNextEncodeOfs, m_uSearchEndOfs, m_uMatchRelaOfs)) == 0 ) { //search matching string
431  encode(*(m_Buf.pRawBuf + m_uNextEncodeOfs), 1); //non-encoding
432  m_uNextEncodeOfs ++;
433  if( is_output_full() ) return 2;
434  continue;
435  }
436  }
437 
438  //maximum length loop
439  do {
440  m_bNeedSearch = true;
441 
442  if( m_uMatchLen == 2 && m_uMatchRelaOfs >= 0x100 ) {
443  encode(*(m_Buf.pRawBuf + m_uNextEncodeOfs), 1); //non-encoding
444  m_uNextEncodeOfs ++;
445  if( is_output_full() ) return 2;
446  break;
447  }
448 
449  if( m_bLongestSearch && (
450  (m_uEndOfSrcFill == 0 && m_uNextEncodeOfs < (m_uEncodeMatchEndOfs - 1))
451  || (m_uEndOfSrcFill != 0 && m_uNextEncodeOfs < (m_uEncodeMatchEndOfs - m_uMatchLen - 1)) //step 4, check if the matching string exceed m_uEncodeMatchEndOfs?
452  )
453  ) {
454  ushort uSaveMatchRelaOfs = m_uMatchRelaOfs;
455  ushort uSaveMatchLen = m_uMatchLen; // save position and length, search maximum length string from next byte
456  if( ((m_uMatchLen = zip_search(m_uNextEncodeOfs + 1, m_uSearchEndOfs, m_uMatchRelaOfs)) > uSaveMatchLen) &&
457  (uSaveMatchLen + 1 < m_uMatchLen || uSaveMatchRelaOfs > 0x80) ) {
458  m_bNeedSearch = false;
459  encode(*(m_Buf.pRawBuf + m_uNextEncodeOfs), 1); //non-encoding
460  m_uNextEncodeOfs ++;
461  if( is_output_full() ) return 2;
462  continue;
463  }
464  encode(uSaveMatchRelaOfs, uSaveMatchLen); //not longer than the first one
465  m_uNextEncodeOfs += uSaveMatchLen;
466  if( is_output_full() ) return 2;
467  break;
468  }
469  //the match string length exceed m_uEncodeMatchEndOfs
470  encode(m_uMatchRelaOfs, m_uMatchLen); //length position code
471  m_uNextEncodeOfs += m_uMatchLen;
472  if( is_output_full() ) return 2;
473  break;
474  } while ( 1 );
475  } while( m_uNextEncodeOfs < m_uEncodeMatchEndOfs );
476 
477  if( m_uEndOfSrcFill != 0 ) break; //end
478  }
479 
480  //end flag
481  encode_end();
482  return 0;
483  }

643  int zip_decode() noexcept
644  {
645  if( !check_flag_buffer() ) return 1;
646  if( is_output_full() ) return 2;
647 
648  do {
649  //InLenBits or InPosBits
650  if( m_step == 0 || m_step == 2 ) {
651  while( 1 ) {
652  //process bit
653  if( m_bit_num != 0 ) {
654  if( m_step == 0 ) {
655  if( m_bit_num == 1 ) {
656  if( m_bit == 0 ) { //non-encoding
657  m_len = 1;
658  m_step = 1; //InLenLen
659  break;
660  }
661  }
662  else if( m_bit_num == 2 ) {
663  m_len = m_bit;
664  }
665  else {
666  m_len = (m_len << 1) | m_bit;
667  if( m_bit_num == 3 && m_len == 0 ) { //read 2 bits, length = 3
668  m_len = g_DecodeLenTable[m_len];
669  m_step = 2; //InPosBits
670  break;
671  }
672  else if( m_bit_num == 4 && m_len <= 5 ) { //read 3 bits, 011,101,100,010, -> 2,4,5,>9
673  m_len = g_DecodeLenTable[m_len];
674  m_step = 1; //InLenLen
675  break;
676  }
677  else if( m_bit_num == 5 ) {
678  m_len = g_DecodeLenTable[m_len];
679  m_step = 1; //InLenLen
680  break;
681  }
682  }
683  }
684  else {
685  if( m_bit_num == 1 ) {
686  if( m_bit != 0 ) {
687  m_pos = 0;
688  m_step = 3; //InPosLen
689  break;
690  }
691  }
692  else if( m_bit_num == 2 ) {
693  m_pos = m_bit;
694  }
695  else {
696  m_pos = (m_pos << 1) | m_bit;
697  if( m_bit_num == 4 && m_pos <= 1 ) { //read 3 bits, 000 or 001, -> 1 or 2 (high byte)
698  m_pos = (ushort)g_DecodePosTable[m_pos] << 8;
699  m_step = 3; //InPosLen
700  break;
701  }
702  else if( m_bit_num == 5 && m_pos <= 7 ) { //read 4 bits, if < 8, -> 3,4,5,6 (high byte)
703  m_pos = (ushort)g_DecodePosTable[m_pos] << 8;
704  m_step = 3; //InPosLen
705  break;
706  }
707  else if( m_bit_num == 6 && m_pos <= 0x16 ) { //read 5 bits
708  m_pos = (ushort)g_DecodePosTable[m_pos] << 8;
709  m_step = 3; //InPosLen
710  break;
711  }
712  else if( m_bit_num == 7 ) { //read 6 bits, remove high bits
713  m_pos = (m_pos & 0xDF) << 8;
714  m_step = 3; //InPosLen
715  break;
716  }
717  }
718  } //end if m_step
719  } //end if m_bit_num != 0
720 
721  //get bit
722  m_bit = get_bit();
723  m_bit_num ++;
724 
725  //process empty flag buffer
726  if( !check_flag_buffer() ) return 1;
727  }
728 
729  m_bit_num = 0; //reset
730  }
731 
732  //bytes
733  if( m_step == 1 ) { //InLenLen
734  if( m_len == 1 ) {
735  if( !check_request_bytes(1) ) return 1;
736  *(m_Buf.pRawBuf + m_Buf.uCodeStoreOfs) = (byte)zip_get_bytes(1);
737  m_Buf.uCodeStoreOfs ++;
738  m_step = 0; //InLenBits
739  if( is_output_full() ) return 2;
740  }
741  else if( m_len == 10 ) { //length>=10, read a byte, v+10
742  if( !check_request_bytes(1) ) return 1;
743  if( (m_len += (ushort)(byte)zip_get_bytes(1)) == 0x109 ) break; //0xFF, end
744  m_step = 2; //InPosBits
745  }
746  else if( m_len == 2 ) {
747  m_pos = 0;
748  m_step = 3; //InPosLen
749  }
750  else {
751  m_step = 2; //InPosBits
752  }
753  }
754  else if( m_step == 3 ) { //InPosLen
755  if( !check_request_bytes(1) ) return 1;
756  m_pos |= (ushort)(byte)zip_get_bytes(1); //low byte
757  ushort i = m_Buf.uCodeStoreOfs - m_pos; //copy string
758  do {
759  *(m_Buf.pRawBuf + m_Buf.uCodeStoreOfs) = *(m_Buf.pRawBuf + i);
760  i ++;
761  m_Buf.uCodeStoreOfs ++;
762  } while( --m_len );
763  m_step = 0; //InLenBits
764  if( is_output_full() ) return 2;
765  }
766  } while( 1 );
767 
768  //end
769  return 0;
770  }
