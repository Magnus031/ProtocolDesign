38 inline uintptr _Compile_One_File(ShareCom<IWlangParser>& spParser, const StringS& strSrc, const StringS& strDest) throw()
39 {
40  CallResult cr;
41 
42  //file name
43  ConsoleHelper::WriteLine(StringUtilHelper::To_ConstString(strSrc));
44 
45  //output
46  ShareCom<ICplMetaData> spMeta;
47  cr = CplMetaDataHelper::CreateCplMetaData(spMeta);
48  if( cr.IsFailed() ) {
49  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Initialization Failed!")));
50  return 1;
51  }
52 
53  //source file
54  {
55  ShareCom<IByteStream> spStream;
56  cr = StreamHelper::CreateFileStream(StringUtilHelper::To_ConstString(strSrc), FileOpenTypes::Read, FileCreationTypes::Open, spStream);
57  if( cr.IsFailed() ) {
58  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: The Source File cannot be opened!")));
59  return 1;
60  }
61  ShareCom<ITextStreamRoot> spText;
62  cr = StreamHelper::CreateTextStream(spText);
63  if( cr.IsFailed() ) {
64  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: The Source File cannot be opened as a text file!")));
65  return 1;
66  }

68  _COMPONENT_INSTANCE_INTERFACE(ITextStreamRoot, ITextUtility, spText, spTU, cr);

70  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: Cannot process the source file!")));
71  return 1;
72  }
73  spTU.Deref().SetStream(spStream);
74  // BOM
75  int iBOMType;
76  cr = spText.Deref().CheckBOM(iBOMType);
77  if( cr.IsFailed() ) {
78  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: The Source File must have a BOM with UTF8!")));
79  return 1;
80  }
81  if( iBOMType != BOMTypes::UTF8 ) {
82  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: The Source File must have a BOM with UTF8!")));
83  return 1;
84  }
85 
86  //parser
87  spParser.Deref().SetInput(spText);
88  spParser.Deref().SetOutput(spMeta);
89  } //end block
90 
91  //loop
92 
93  cr = spParser.Deref().Start();
94  if( cr.IsFailed() ) {
95  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: The parser cannot start!")));
96  return 1;
97  }
98 
99  while( true ) {
100  cr = spParser.Deref().Parse();
101  if( cr.IsFailed() ) {
102  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: Parsing Failed!")));
103  break;
104  }
105  if( cr.GetResult() == SystemCallResults::S_False )
106  break;
107  thread_sleep(1);
108  }
109 
110  //errors
111  ShareArray<StringS> arrError(spParser.Deref().get_ErrorArray());
112  uintptr uCount = arrError.GetCount();
113  if( uCount != 0 ) {
114  uint uAttrs = STDOUT_ATTR_FORE_RED | STDOUT_ATTR_BACK_KEEP;
115  auto iter(arrError.GetBegin());
116  for( ; iter != arrError.GetEnd(); iter.MoveNext() ) {
117  ConsoleHelper::WriteLine(uAttrs, StringUtilHelper::To_ConstString(iter.get_Value()));
118  }
119  return uCount;
120  }
121  if( cr.IsFailed() )
122  return 1;
123 
124  //save
125  {
126  ShareCom<IByteStream> spStream;
127  cr = StreamHelper::CreateFileStream(StringUtilHelper::To_ConstString(strDest), FileOpenTypes::Write, FileCreationTypes::Create, spStream);
128  if( cr.IsFailed() ) {
129  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: The Destination File cannot be opened!")));
130  return 1;
131  }
132  cr = spMeta.Deref().Save(spStream);
133  if( cr.IsFailed() ) {
134  spStream.Release();
135  _Delete_Output_File(strDest);
136  //saving failed
137  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Error: Cannot save to the Destination File!")));
138  return 1;
139  }
140  } //end block
141 
142  return 0;
143 }
