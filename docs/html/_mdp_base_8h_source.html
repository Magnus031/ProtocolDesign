143  //load project file with WON format
144  bool Load(const ConstStringS& strFile)
145  {
146  CallResult cr;
147 
148  ShareCom<ITextStreamRoot> spText;
149  {
150  ShareCom<IByteStream> spStream;
151  cr = StreamHelper::CreateFileStream(strFile, FileOpenTypes::Read, FileCreationTypes::Open, spStream);
152  if( cr.IsFailed() )
153  return false;
154  cr = StreamHelper::CreateTextStream(spText);
155  if( cr.IsFailed() )
156  return false;
157  ShareCom<ITextUtility> spTU;
158  _COMPONENT_INSTANCE_INTERFACE(ITextStreamRoot, ITextUtility, spText, spTU, cr);
159  if( cr.IsFailed() )
160  return false;
161  spTU.Deref().SetStream(spStream);
162  // BOM
163  int iType;
164  cr = spText.Deref().CheckBOM(iType);
165  if( cr.IsFailed() )
166  return false;
167  if( iType != BOMTypes::UTF8 )
168  return false;
169  } //end block
170 
171  ShareCom<IWonParser> spParser;
172  {
173  ShareCom<IWonUtility> spU;
174  cr = ParserHelper::CreateWonUtility(spU);
175  if( cr.IsFailed() )
176  return false;
177  cr = spU.Deref().Initialize();
178  if( cr.IsFailed() )
179  return false;
180  cr = ParserHelper::CreateWonParser(spParser);
181  if( cr.IsFailed() )
182  return false;
183  cr = spParser.Deref().Initialize(spU);
184  if( cr.IsFailed() )
185  return false;
186  } //end block
187 
188  spParser.Deref().SetInput(spText);
189  cr = spParser.Deref().Start();
190  if( cr.IsFailed() )
191  return false;
192 
193  StringA strToken;
194  RefPtr<StringA> refString;
195  fileTreeClass::Iterator iterParent;
196  fileTreeClass::Iterator iterAfter;
197 
198  m_iterTitle = fileTreeClass::Iterator();
199  m_iterContent = fileTreeClass::Iterator();
200  m_iterCopyright = fileTreeClass::Iterator();
201 
202  int iState = STATE_START;
203  //loop
204  while( true ) {
205  int iType;
206  cr = spParser.Deref().GetNextToken(iType, strToken);
207  if( cr.IsFailed() )
208  return false;
209  if( cr.GetResult() == SystemCallResults::S_False )
210  break;
211  //types
212  switch( iType ) {
213  case WonTokenTypes::BeginObject:
214  if( iState == STATE_START ) {
215  iState = STATE_BASE;
216  }
217  else if( iState == STATE_TREE ) {
218  iterAfter = m_tree.Insert(iterParent, iterAfter); //may throw
219  }
220  else {
221  return false;
222  }
223  break;
224  case WonTokenTypes::EndObject:
225  if( iState == STATE_BASE ) {
226  iState = STATE_START;
227  }
228  else if( iState == STATE_TREE ) {
229  ;
230  }
231  else {
232  return false;
233  }
234  break;
235  case WonTokenTypes::BeginArray:
236  if( iState == STATE_COMMANDS ) {
237  ;
238  }
239  else if( iState == STATE_TREE ) {
240  ;
241  }
242  else {
243  return false;
244  }
245  break;
246  case WonTokenTypes::EndArray:
247  if( iState == STATE_COMMANDS ) {
248  ;
249  }
250  else if( iState == STATE_TREE ) {
251  ;
252  }
253  else {
254  return false;
255  }
256  break;
257  case WonTokenTypes::Key:
258  if( iState == STATE_BASE ) {
259  if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"project-name\"")) ) {
260  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strProjectName);
261  }
262  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"document-directory\"")) ) {
263  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strDocumentDirectory);
264  }
265  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"language\"")) ) {
266  iState = STATE_LANG;
267  }
268  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"command\"")) ) {
269  if( m_arrCommand.IsBlockNull() )
270  m_arrCommand = ShareArrayHelper::MakeShareArray<StringA>(MemoryHelper::GetCrtMemoryManager()); //may throw
271  m_arrCommand.RemoveAll();
272  iState = STATE_COMMANDS;
273  }
274  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"cover\"")) ) {
275  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strCoverImageFile);
276  }
277  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"cover-name\"")) ) {
278  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strCoverName);
279  }
280  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"topic\"")) ) {
281  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strTopic);
282  }
283  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"identifier\"")) ) {
284  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strIdentifier);
285  }
286  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"tree\"")) ) {
287  m_tree.RemoveAll();
288  iState = STATE_TREE;
289  }
290  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"author\"")) ) {
291  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strAuthor);
292  }
293  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"description\"")) ) {
294  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strDescription);
295  }
296  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"date\"")) ) {
297  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strDate);
298  }
299  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"contributor\"")) ) {
300  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strContributor);
301  }
302  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"publisher\"")) ) {
303  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strPublisher);
304  }
305  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"subject\"")) ) {
306  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strSubject);
307  }
308  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"rights\"")) ) {
309  refString = RefPtrHelper::MakeRefPtr<StringA>(m_strRights);
310  }
311  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"reading-order\"")) ) {
312  iState = STATE_READING_ORDER;
313  }
314  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"line-layout\"")) ) {
315  iState = STATE_LINE_LAYOUT;
316  }
317  else {
318  return false;
319  }
320  }
321  else if( iState == STATE_TREE ) {
322  if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"tree\"")) ) {
323  //enter the next level
324  iterParent = iterAfter;
325  iterAfter = fileTreeClass::Iterator();
326  }
327  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"name\"")) ) {
328  refString = RefPtrHelper::MakeRefPtr<StringA>(iterAfter.get_Value().get_First());
329  iState = STATE_TREE_IN_NODE;
330  }
331  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"file\"")) ) {
332  refString = RefPtrHelper::MakeRefPtr<StringA>(iterAfter.get_Value().get_Second());
333  iState = STATE_TREE_IN_NODE;
334  }
335  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "\"guide\"")) ) {
336  iState = STATE_TREE_IN_GUIDE;
337  }
338  else {
339  return false;
340  }
341  }
342  else {
343  return false;
344  }
345  break;
346  case WonTokenTypes::EndPair:
347  if( iState == STATE_BASE ) {
348  ;
349  }
350  else if( iState == STATE_COMMANDS ) {
351  iState = STATE_BASE;
352  }
353  else if( iState == STATE_TREE_IN_NODE ) {
354  iState = STATE_TREE;
355  }
356  else if( iState == STATE_TREE_IN_GUIDE ) {
357  iState = STATE_TREE;
358  }
359  else if( iState == STATE_TREE ) {
360  if( !iterParent.IsNull() && iterParent != m_tree.GetRoot() ) {
361  //back to parent
362  iterAfter = iterParent;
363  iterParent = iterAfter;
364  iterParent.MoveParent();
365  }
366  else {
367  iState = STATE_BASE;
368  }
369  }
370  else {
371  return false;
372  }
373  break;
374  case WonTokenTypes::ValueString:
375  delete_quotes(strToken);
376  if( iState == STATE_BASE ) {
377  refString.Deref() = strToken;
378  }
379  else if( iState == STATE_TREE_IN_NODE ) {
380  refString.Deref() = strToken;
381  }
382  else if( iState == STATE_TREE_IN_GUIDE ) {
383  if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "title")) ) {
384  if( !m_iterTitle.IsNull() )
385  return false;
386  m_iterTitle = iterAfter;
387  }
388  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "content")) ) {
389  if( !m_iterContent.IsNull() )
390  return false;
391  m_iterContent = iterAfter;
392  }
393  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "copyright")) ) {
394  if( !m_iterCopyright.IsNull() )
395  return false;
396  m_iterCopyright = iterAfter;
397  }
398  else {
399  return false;
400  }
401  }
402  else if( iState == STATE_COMMANDS ) {
403  m_arrCommand.Add(strToken); //may throw
404  }
405  else if( iState == STATE_READING_ORDER ) {
406  if( !strToken.IsEmpty() ) {
407  if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "rtl")) ) {
408  m_bRTLorder = true;
409  }
410  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "ltr")) ) {
411  m_bRTLorder = false;
412  }
413  else {
414  return false;
415  }
416  }
417  iState = STATE_BASE;
418  }
419  else if( iState == STATE_LINE_LAYOUT ) {
420  if( !strToken.IsEmpty() ) {
421  if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "v")) ) {
422  m_bVerticalLine = true;
423  }
424  else if( ConstStringCompareTrait<ConstStringA>::IsEQ(StringUtilHelper::To_ConstString(strToken), DECLARE_TEMP_CONST_STRING(ConstStringA, "h")) ) {
425  m_bVerticalLine = false;
426  }
427  else {
428  return false;
429  }
430  }
431  iState = STATE_BASE;
432  }
433  else {
434  return false;
435  }
436  break;
437  case WonTokenTypes::ValueNumberInteger:
438  if( iState == STATE_LANG ) {
439  bool bOK;
440  string_to_value(ShareArrayHelper::GetInternalPointer(strToken), 10, m_uLCID, bOK); //no check
441  (void)bOK;
442  assert( bOK );
443  if( m_uLCID == 0 )
444  m_uLCID = 1033;
445  iState = STATE_BASE;
446  }
447  else {
448  return false;
449  }
450  break;
451  default:
452  return false;
453  break;
454  } //end switch
455  } //end while
456 
457  assert( iState == STATE_START );
458 
459  return true;
460  }

512 class FileTreeEnumerator
513 {
514 public:
515  FileTreeEnumerator(const fileTreeClass& tree) throw() : m_tree(tree), m_bHaveChild(false), m_uLevel(0), m_uDelta(0), m_bNegative(false)
516  {
517  }
518  ~FileTreeEnumerator() throw()
519  {
520  }
521 
522  bool FindFirst() throw()
523  {
524  m_bHaveChild = false;
525  m_uLevel = 1;
526  m_uDelta = 0;
527  m_bNegative = false;
528  m_iter = m_tree.GetRoot();
529  if( m_iter.IsNull() )
530  return false;
531  m_iter.MoveChild();
532  if( m_iter.IsNull() )
533  return false;
534  check_child();
535  return true;
536  }
537  bool FindNext() throw()
538  {
539  auto iter(m_iter);
540  m_bNegative = false;
541  //child
542  iter.MoveChild();
543  if( !iter.IsNull() ) {
544  m_iter = iter;
545  check_child();
546  m_uLevel ++;
547  m_uDelta = 1;
548  return true;
549  }
550  //next
551  iter = m_iter;
552  iter.MoveNext();
553  if( !iter.IsNull() ) {
554  m_iter = iter;
555  check_child();
556  m_uDelta = 0;
557  return true;
558  }
559  //parent
560  iter = m_iter;
561  m_uDelta = 0;
562  m_bNegative = true;
563  while( true ) {
564  iter.MoveParent();
565  m_uLevel --;
566  m_uDelta ++;
567  if( iter.IsNull() || iter == m_tree.GetRoot() ) {
568  -- m_uDelta;
569  break;
570  }
571  iter.MoveNext();
572  if( !iter.IsNull() ) {
573  m_iter = iter;
574  check_child();
575  return true;
576  }
577  } //end while
578  if( m_uDelta == 0 )
579  m_bNegative = false;
580  return false;
581  }
582 
583  ConstStringA GetName() const throw()
584  {
585  return StringUtilHelper::To_ConstString(m_iter.get_Value().get_First());
586  }
587  ConstStringA GetFile() const throw()
588  {
589  return StringUtilHelper::To_ConstString(GetFileString());
590  }
591  StringA GetFileString() const throw()
592  {
593  return m_iter.get_Value().get_Second();
594  }
595  bool IsLeaf() const throw()
596  {
597  return !m_bHaveChild;
598  }
599  uintptr GetLevel() const throw()
600  {
601  return m_uLevel;
602  }
603  uintptr GetDelta(bool& bNegative) const throw()
604  {
605  bNegative = m_bNegative;
606  return m_uDelta;
607  }
608 
609 private:
610  void check_child() throw()
611  {
612  auto iter(m_iter);
613  iter.MoveChild();
614  m_bHaveChild = !iter.IsNull();
615  }
616 
617 private:
618  const fileTreeClass& m_tree;
619  fileTreeClass::Iterator m_iter;
620  bool m_bHaveChild;
621  uintptr m_uLevel;
622  uintptr m_uDelta;
623  bool m_bNegative;
624 
625 private:
626  //noncopyable
627  FileTreeEnumerator(const FileTreeEnumerator&) throw();
628  FileTreeEnumerator& operator=(const FileTreeEnumerator&) throw();
629 };

633 template <class TString>
634 class FileListT
635 {
636 private:
637  typedef HashList<TString, StringHashTrait<TString>, StringCompareTrait<TString>> HashListClass;
638 
639 public:
640  FileListT() throw() : m_hashList(MemoryHelper::GetCrtMemoryManager())
641  {
642  }
643  ~FileListT() throw()
644  {
645  }
646 
647  uintptr GetCount() const throw()
648  {
649  return m_arrFile.GetCount();
650  }
651  const TString& GetAt(uintptr uIndex) const throw()
652  {
653  assert( uIndex < GetCount() );
654  return m_arrFile.GetAt(uIndex).get_Value();
655  }
656  bool Find(const TString& str) const throw()
657  {
658  return m_hashList.Find(str) != m_hashList.GetEnd();
659  }
660 
661  void Clear() throw()
662  {
663  m_hashList.RemoveAll();
664  if( !m_arrFile.IsBlockNull() )
665  m_arrFile.RemoveAll();
666  }
667  //add a file
668  void AddFile(const TString& str)
669  {
670  if( Find(str) )
671  return ;
672  auto iter(m_hashList.InsertWithoutFind(str)); //may throw
673  try {
674  if( m_arrFile.IsBlockNull() )
675  m_arrFile = ShareArrayHelper::MakeShareArray<typename HashListClass::Position>(MemoryHelper::GetCrtMemoryManager()); //may throw
676  m_arrFile.Add(iter.GetPosition()); //may throw
677  }
678  catch(...) {
679  m_hashList.RemoveAt(iter);
680  throw; //re-throw
681  }
682  }
683 
684 private:
685  HashListClass m_hashList;
686  ShareArray<typename HashListClass::Position> m_arrFile;
687 
688 private:
