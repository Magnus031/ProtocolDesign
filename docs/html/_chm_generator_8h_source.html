119 inline bool _Generate_Content_File(const ConstStringS& strFile,
120  const ConstStringA& strProjectName,
121  const ConstStringA& strCoverName,
122  FileTreeEnumerator& ftEnum)
123 {
124  //content
125  StringA strContent(StringHelper::MakeEmptyString<CharA>(MemoryHelper::GetCrtMemoryManager())); //may throw
126  //tree
127  StringA strTree(StringHelper::MakeEmptyString<CharA>(MemoryHelper::GetCrtMemoryManager())); //may throw
128  //cover
129  _Generate_Cover_LI_String(strProjectName, strCoverName, strContent); //may throw
130  StringUtilHelper::MakeString(StringUtilHelper::To_ConstString(strContent), strTree); //may throw
131  //files
132  bool bFound = ftEnum.FindFirst();
133  while( bFound ) {
134  //level
135  bool bNegative;
136  uintptr uDelta = ftEnum.GetDelta(bNegative);
137  if( bNegative ) {
138  //leave
139  for( uintptr i = 0; i < uDelta; i ++ )
140  StringUtilHelper::Append(DECLARE_TEMP_CONST_STRING(ConstStringA, "</UL>\r\n"), strTree); //may throw
141  }
142  else if( uDelta > 0 ) {
143  //enter
144  StringUtilHelper::Append(DECLARE_TEMP_CONST_STRING(ConstStringA, "<UL>\r\n"), strTree); //may throw
145  } //end if
146  //node
147  _Generate_Htm_LI_String(ftEnum.GetName(), ftEnum.GetFile(), ftEnum.IsLeaf(), strContent); //may throw
148  StringUtilHelper::Append(strContent, strTree); //may throw
149  bFound = ftEnum.FindNext();
150  } //end while
151  bool bNegative;
152  uintptr uDelta = ftEnum.GetDelta(bNegative);
153  if( bNegative ) {
154  //leave
155  for( uintptr i = 0; i < uDelta; i ++ )
156  StringUtilHelper::Append(DECLARE_TEMP_CONST_STRING(ConstStringA, "</UL>\r\n"), strTree); //may throw
157  } //end if
158  //content
159  StringUtilHelper::MakeString(ConstStringA(g_chm_content_body::GetAddress(), g_chm_content_body::GetCount()), strContent); //may throw
160  StringUtilHelper::Replace(DECLARE_TEMP_CONST_STRING(ConstStringA, "$$GENERATOR$$"), get_chm_generator(), strContent); //may throw
161  StringUtilHelper::Replace(DECLARE_TEMP_CONST_STRING(ConstStringA, "$$TREE$$"), StringUtilHelper::To_ConstString(strTree), strContent); //may throw
162  return _Generate_Fix_Content_File(strFile, StringUtilHelper::To_ConstString(strContent));
163 }
