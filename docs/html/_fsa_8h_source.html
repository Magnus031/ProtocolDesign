201  void ProcessState(uint uEventNo) throw()
202  {
203  assert( !_Is_Fsa_Table_Null(m_table) );
204  assert( m_iCurrentState > 0 && m_iCurrentState <= m_table.iMaxStateNo );
205  m_iPrevState = m_iCurrentState;
206  m_uLastEventNo = uEventNo;
207  //look table
208  const FSA_TRANSITION_ITEM* pItem = m_table.pState[m_iCurrentState].pTransition;
209  if( pItem == NULL ) {
210  m_iCurrentState = m_table.pState[m_iCurrentState].iDefaultState;
211  post_process_current_state(uEventNo);
212  return ;
213  }
214  while( pItem->uEventFirstNo != FSA_LAST_EVENT_NO ) {
215  if( uEventNo >= pItem->uEventFirstNo && uEventNo <= pItem->uEventLastNo ) {
216  m_iCurrentState = pItem->iNextState;
217  post_process_current_state(uEventNo);
218  return ;
219  }
220  pItem ++;
221  } //end while
222  m_iCurrentState = m_table.pState[m_iCurrentState].iDefaultState;
223  post_process_current_state(uEventNo);
224  }

329  void post_process_current_state(uint uEventNo) throw()
330  {
331  if( uEventNo != FSA_END_OF_EVENT ) {
332  CallResult cr = SafeOperators::Add(m_uBackEventNum, (uint)1, m_uBackEventNum);
333  if( cr.IsFailed() ) {
334  //overflow
335  m_iCurrentState = 0;
336  m_iLastStopState = 1;
337  }
338  }
339  if( m_iCurrentState > 0 ) {
340  assert( uEventNo != FSA_END_OF_EVENT );
341  int iMatchIndex = m_table.pState[m_iCurrentState].iMatchIndex;
342  if( iMatchIndex <= 0 ) {
343  //matched stop state
344  bool bUnput;
345  m_iMatch = get_match(iMatchIndex, bUnput);
346  //stop state
347  m_iLastStopState = iMatchIndex;
348  m_uBackEventNum = bUnput ? 1 : 0;
349  }
350  }
351  else {
352  //mismatch
353  if( m_iLastStopState > 0 ) {
354  //invalid
355  m_iMatch = 0;
356  m_uBackEventNum = 0;
357  }
358  //otherwise, stop
359  } //end if
360  }
