92 //generate sequence string
93 inline bool generate_console_sequence_string(uint uAttrs, bool bLN, const char_s* sz, uintptr uLength, unique_string_s& str) throw()
94 {
95  call_result cr;
96  str.DeleteAll();
97  //start
98  bool bUnderScore = (uAttrs & STDOUT_ATTR_UNDERSCORE) != 0;
99  char_s szStart[256];
100  int ret = value_to_string(szStart, sizeof(szStart) / sizeof(char_s),
101  _S("\033[%d;%d;%dm"),
102  (uAttrs & STDOUT_ATTR_REVERSE) ? (7) : (bUnderScore ? (4) : ((uAttrs & (STDOUT_ATTR_FORE_INTENSITY | STDOUT_ATTR_BACK_INTENSITY)) ? (1) : (0))), //mode
103  (uAttrs & STDOUT_ATTR_FORE_KEEP) ? (bUnderScore ? 38 : 39) : (uAttrs & (STDOUT_ATTR_FORE_RED | STDOUT_ATTR_FORE_GREEN | STDOUT_ATTR_FORE_BLUE)) + 30, //foreground
104  (uAttrs & STDOUT_ATTR_BACK_KEEP) ? (49) : ((uAttrs & (STDOUT_ATTR_BACK_RED | STDOUT_ATTR_BACK_GREEN | STDOUT_ATTR_BACK_BLUE)) >> 4) + 40 //background
105  );
106  if( ret > 0 ) {
107  cr = str.AppendNormal(ret, szStart);
108  if( cr.IsFailed() )
109  return false;
110  }
111  //middle
112  if( uLength > 0 ) {
113  cr = str.AppendNormal(uLength, sz);
114  if( cr.IsFailed() )
115  return false;
116  }
117  //tail
118  DECLARE_LOCAL_STATIC_CONST_STRING(char_s, c_szTail, c_uTailLen, _S("\033[0m"))
119  cr = str.AppendNormal(c_uTailLen, c_szTail);
120  if( cr.IsFailed() )
121  return false;
122  //LN
123  if( bLN ) {
124  cr = str.AppendNormal(1, _S('\n'));
125  if( cr.IsFailed() )
126  return false;
127  }
128  return true;
129 }
