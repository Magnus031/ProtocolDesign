113 inline int compare_string(const char_h* s1, const char_h* s2) throw()
114 {
115  const char_h* p1 = s1;
116  const char_h* p2 = s2;
117  int n = 0;
118  do {
119  n = (int)(*p1) - (int)(*p2);
120  if( n != 0 )
121  break;
122  if( (*p1 == 0) || (*p2 == 0) )
123  break;
124  ++ p1; ++ p2;
125  } while( 1 );
126  return n;
127 }

176 // uSize : buffer size in typed characters.
177 // return value : the number of typed characters, <0 means fail.
178 inline int value_to_string(char_a* szBuffer, uintptr uSize, const char_a* szFormat, ...) throw()
179 {
180  assert( uSize > 0 );
181  va_list ap;
182  va_start(ap, szFormat);
183  int ret = ::vsnprintf(szBuffer, uSize, szFormat, ap);
184  va_end(ap);
185  //check
186  ret = (ret < 0) ? INT_FLAG_SIGN_BIT : (((uintptr)ret) >= uSize) ? (int)(uSize - 1) : ret;
187  szBuffer[VTOS_LENGTH_FROM_RETURN(ret)] = '\0';
188  return ret;
189 }
