210  }
211  _CS_CVT_O(thisClass&& src) throw() : baseClass(rv_forward(static_cast<baseClass&>(src))),

222  m_strC = rv_forward(src.m_strC);
223  return *this;
224  }

226  //the return value may indicate the same object when initialized by StringT or UniqueStringT
227  // this method can be called only once.
228  typename TCCTrait::Estring2&& GetV() //may throw
229  {
230  StringOpHelper::Initialize(baseClass::m_str); //may throw or not
231  if( !m_strC.IsNull() && m_strC.GetCount() != 0 )
232  StringUtilHelper::MakeString(m_strC, baseClass::m_str); //may throw
233  return baseClass::GetV();
234  }
235  //DO NOT assign the return value to other variables
236  // this method can be called many times when using class object.
237  ConstStringT<typename TCCTrait::Estring2::EType> GetC() const throw()
238  {
239  return m_strC.IsNull() ? StringUtilHelper::To_ConstString(baseClass::m_str) : m_strC;
240  }

242  template <typename... Args>
243  void Init(const ConstStringT<typename TCCTrait::Echar1>& str, Args&&... args) //may throw
244  {
245  if( TCCTrait::c_bSame ) {
246  StringOpHelper::Uninitialize(baseClass::m_str);
247  ConstArrayHelper::SetInternalPointer<typename TCCTrait::Estring2::EType>((const typename TCCTrait::Estring2::EType*)ConstArrayHelper::GetInternalPointer<typename TCCTrait::Echar1>(str), str.GetCount(), m_strC);
248  return ;
249  }
250  baseClass::cvt_string(str, rv_forward<Args>(args)...);
251  ConstArrayHelper::SetInternalPointer<typename TCCTrait::Estring2::EType>(NULL, 0, m_strC);
252  }
253  template <typename... Args>
254  void Init(typename TCCTrait::Estring2::template StringType<typename TCCTrait::Echar1>&& str, Args&&... args) //may throw
255  {
256  if( TCCTrait::c_bSame )
257  baseClass::m_str = rv_forward(*((typename TCCTrait::Estring2*)(&str)));
258  else
259  baseClass::cvt_string(StringUtilHelper::To_ConstString(str), rv_forward<Args>(args)...);
260  ConstArrayHelper::SetInternalPointer<typename TCCTrait::Estring2::EType>(NULL, 0, m_strC);
261  }
