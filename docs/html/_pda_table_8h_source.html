131  //state: 2 ~ iMaxStateNo
132  //transition: the count of arrTransitionNum is iMaxStateNo + 1
133  //rule: 0 ~ iMaxRuleNo
134  void Allocate(int iMaxStateNo, const ConstArray<int>& arrTransitionNum, int iMaxRuleNo)
135  {
136  assert( _Is_Pda_Size_Valid(iMaxStateNo, iMaxRuleNo) );
137  assert( arrTransitionNum.GetCount() == (uintptr)(iMaxStateNo + 1) );
138  assert( _Is_Pda_Table_Null(m_table) );
139  //transition number
140  int total_transition_num = 0;
141  for( int i = 2; i <= iMaxStateNo; i ++ ) {
142  int iNum = arrTransitionNum[i];
143  assert( iNum > 0 );
144  iNum = SafeOperators::AddThrow(iNum, (int)1); //may throw
145  total_transition_num = SafeOperators::AddThrow(total_transition_num, iNum); //may throw
146  }
147  //allocate
148  uint uBytes, uBytes1, uBytes2;
149  uBytes1 = SafeOperators::MultiplyThrow((uint)iMaxStateNo + 2, (uint)sizeof(PDA_STATE_ITEM)); //may throw
150  uBytes2 = SafeOperators::MultiplyThrow((uint)iMaxRuleNo + 2, (uint)sizeof(PDA_RULE_ITEM)); //may throw
151  uBytes = SafeOperators::AddThrow(uBytes1, uBytes2); //may throw
152  uBytes1 = SafeOperators::MultiplyThrow((uint)total_transition_num, (uint)sizeof(PDA_TRANSITION_ITEM)); //may throw
153  uBytes = SafeOperators::AddThrow(uBytes, uBytes1); //may throw
154  uint uStart = m_allocator.Deref().Allocate(uBytes);
155  if( uStart == 0 )
156  throw OutOfMemoryException();
157  //pointers
158  PDA_STATE_ITEM* pState = (PDA_STATE_ITEM*)(m_allocator.Deref().ToPointer(uStart));
159  pState[0].iDefaultState = 0;
160  pState[0].pTransition = NULL;
161  pState[1].iDefaultState = 0;
162  pState[1].pTransition = NULL;
163  pState[iMaxStateNo + 1].iDefaultState = 0;
164  pState[iMaxStateNo + 1].pTransition = NULL;
165  PDA_RULE_ITEM* pRule = (PDA_RULE_ITEM*)((byte*)pState + sizeof(PDA_STATE_ITEM) * (iMaxStateNo + 2));
166  pRule[0].uLeftEventNo = 0;
167  pRule[0].uRightSymbolNumber = 1;
168  pRule[iMaxRuleNo + 1].uLeftEventNo = 0;
169  pRule[iMaxRuleNo + 1].uRightSymbolNumber = 0;
170  PDA_TRANSITION_ITEM* pItem = (PDA_TRANSITION_ITEM*)((byte*)pRule + sizeof(PDA_RULE_ITEM) * (iMaxRuleNo + 2));
171  for( int i = 2; i <= iMaxStateNo; i ++ ) {
172  pState[i].pTransition = pItem;
173  pItem += (arrTransitionNum[i]);
174  pItem->uEventFirstNo = PDA_LAST_EVENT_NO;
175  pItem->uEventLastNo = PDA_LAST_EVENT_NO;
176  pItem->iNextState = 0;
177  pItem ++;
178  }
179  //parameters
180  _Init_Pda_Table(m_table, iMaxStateNo, pState, iMaxRuleNo, pRule);
181  }
