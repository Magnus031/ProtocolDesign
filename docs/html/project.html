Project Types

Three types of project are supported by GKC/GDAM system.

Type Description Windows Linux
shared Shared Assembly DLL SO
console Console application Console sub-system Foreground process
normal Normal application Windows sub-system Foreground/Backgroud process

Basic Settings

User projects which use GKC should include GKC.cmk file which provides the imported target: GKC::Sys. The content of top-level CMakeLists.txt is listed as follows:

cmake_minimum_required(VERSION 3.16)
project(Prj1)
include("$ENV{GDAM_SYSTEM}/dev/GKC.cmk")
# target directories
add_subdirectory(dir1 dir1)
add_subdirectory(dir2 dir2)
...

The content of CMakeLists.txt under sub-directory for target is shown as follows:

# example 1
add_executable(t1 "src/Main.cpp")
target_include_directories(t1 PRIVATE
"${CMAKE_CURRENT_BINARY_DIR}"
"${CMAKE_CURRENT_SOURCE_DIR}/include"
...
)
target_precompile_headers(t1 PRIVATE "include/PreComp.h")
TARGET_SET_TYPE(t1 "console")
set_target_properties(t1 PROPERTIES
RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin"
)
target_link_libraries(t1 GKC::Sys)
# example 2
add_executable(t2 "src/Main.cpp")
target_include_directories(t2 PRIVATE
"${CMAKE_CURRENT_BINARY_DIR}"
"${CMAKE_CURRENT_SOURCE_DIR}/include"
...
)
target_precompile_headers(t2 PRIVATE "PreComp.h")
TARGET_SET_TYPE(t2 "normal")
target_link_libraries(t2 GKC::Sys)
# example 3
add_library(t3 SHARED "src/EntryPoint.cpp")
target_include_directories(t3 PRIVATE
"${CMAKE_CURRENT_BINARY_DIR}"
"${CMAKE_CURRENT_SOURCE_DIR}/include"
...
)
target_precompile_headers(t3 PRIVATE "PreComp.h")
TARGET_SET_TYPE(t3 "shared")
set_target_properties(t3 PROPERTIES
RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin"
)
target_link_libraries(t3 GKC::Sys)

If user projects want to use other CPKG packages, they should include cpkg.cmk file which also includes GKC.cmk file. An example is listed as follows:

cmake_minimum_required(VERSION 3.16)
project(Prj1)
include("$ENV{GDAM_SYSTEM}/dev/cpkg.cmk")
# import packages
CPKG_IMPORT(pkg1)
# target directories
add_subdirectory(dir1 dir1)
add_subdirectory(dir2 dir2)
...
