39  void DoThreadProc() noexcept
40  {
41  //this thread is running
42  atomic_compare_exchange(*((int*)&m_finish), 0, 1);
43 
44  //wait
45  struct epoll_event events[1024];
46  int iTimeout = -1;
47  while ( true ) {
48  //epoll
49  int ret = ::epoll_wait((int)(m_t.m_hdEpoll.GetHandle()), events, (int)(sizeof(events) / sizeof(struct epoll_event)), iTimeout); //may block
50  if ( ret == -1 ) {
51  if ( errno == EINTR )
52  continue;
53  //sleep
54  thread_sleep(1); //1ms
55  continue;
56  }
57  //timeout
58  if ( ret == 0 ) {
59  iTimeout = m_t.DoTimeout(iTimeout);
60  continue;
61  }
62  //events
63  bool bEnd = false;
64  int j = 0;
65  for ( int i = 0; i < ret; ++ i ) {
66  void* p = events[i].data.ptr;
67  //event
68  if ( (uintptr)p == INVALID_ARRAY_INDEX ) {
69  if ( events[i].events & EPOLLIN ) {
70  if ( atomic_compare_exchange((int&)m_exit, 1, 0) )
71  bEnd = true;
72  uint64_t v;
73  ssize_t sret = ::read((int)(m_t.m_hdEvent.GetHandle()), &v, sizeof(uint64_t));
74  (void)sret;
75  assert( sret == sizeof(uint64_t) );
76  call_result cr = epoll_rev_event((int)(m_t.m_hdEpoll.GetHandle()), (int)(m_t.m_hdEvent.GetHandle()), (void*)INVALID_ARRAY_INDEX, false);
77  (void)cr;
78  assert( cr.IsSucceeded() );
79  if ( !bEnd )
80  m_t.DoEvent();
81  }
82  continue;
83  }
84  //process one
85  if ( !m_t.DoItemPre(events[i].events, p) ) {
86  if ( j != i )
87  events[j] = events[i];
88  ++ j;
89  }
90  }
91  //post-processing
92  for ( int i = 0; i < j; ++ i )
93  m_t.DoItemPost(events[i].events, events[i].data.ptr);
94  if ( bEnd )
95  break;
96  iTimeout = m_t.DoAfterEvents(iTimeout);
97  }
98 
99  //end
100  m_p = this;
101  atomic_compare_exchange(*((int*)&m_finish), 0, 1);
102  }
