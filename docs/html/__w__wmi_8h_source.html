42  HRESULT Initialize(LPCOLESTR szNamespace) noexcept
43  {
44  assert( m_spLoc.IsNull() );
45  assert( m_spSvc.IsNull() );
46 
47  // Obtain the initial locator to WMI
48  HRESULT hr = m_spLoc.CoCreateInstance(CLSID_WbemLocator, NULL, CLSCTX_INPROC_SERVER);
49  if ( FAILED(hr) )
50  return hr;
51 
52  // Connect to WMI through the IWbemLocator::ConnectServer method
53 
54  // Connect to the namespace (such as root\cimv2) with
55  // the current user and obtain pointer pSvc
56  // to make IWbemServices calls.
57  _os_com_bstr bstr;
58  if ( !bstr.CopyFrom(szNamespace) ) {
59  m_spLoc.Release();
60  return E_OUTOFMEMORY;
61  }
62  hr = m_spLoc->ConnectServer(bstr, NULL, NULL, NULL, 0,
63  NULL, NULL, &m_spSvc);
64  if ( FAILED(hr) ) {
65  m_spLoc.Release();
66  return hr;
67  }
68 
69  // Set security levels on the proxy
70  hr = ::CoSetProxyBlanket(m_spSvc, RPC_C_AUTHN_WINNT, RPC_C_AUTHZ_NONE, NULL,
71  RPC_C_AUTHN_LEVEL_CALL, RPC_C_IMP_LEVEL_IMPERSONATE,
72  NULL, EOAC_NONE);
73  if ( FAILED(hr) ) {
74  m_spSvc.Release();
75  m_spLoc.Release();
76  return hr;
77  }
78 
79  return hr;
80  }

83  bool EnumFirst(LPCOLESTR szClass, _os_com_ptr<IWbemClassObject>& sp) noexcept
84  {
85  assert( !m_spSvc.IsNull() );
86 
87  HRESULT hr = S_OK;
88 
89  _os_com_bstr bstr1, bstr2;
90  if ( !bstr2.CopyFrom(szClass) )
91  return false;
92  if ( !bstr1.CopyFrom(L"SELECT * FROM ") )
93  return false;
94  hr = bstr1.Append(bstr2);
95  if ( FAILED(hr) )
96  return false;
97  bstr2.Empty();
98  if ( !bstr2.CopyFrom(L"WQL") )
99  return false;
100 
101  // Use the IWbemServices pointer to make requests of WMI
102  IEnumWbemClassObject* pEnum = NULL;
103  hr = m_spSvc->ExecQuery(bstr2, bstr1,
104  WBEM_FLAG_FORWARD_ONLY | WBEM_FLAG_RETURN_IMMEDIATELY,
105  NULL, &pEnum);
106  if ( FAILED(hr) )
107  return false;
108  if ( pEnum == NULL )
109  return false;
110  m_spEnum.Attach(pEnum);
111  return EnumNext(sp);
112  }

119  if ( FAILED(hr) )
120  return false;

123  sp.Attach(pObj);
