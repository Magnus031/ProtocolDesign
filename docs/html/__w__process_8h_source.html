405  BOOL Create(LPCWSTR szCmdLine, LPCWSTR szWorkPath = NULL,
406  DWORD dwFlags = 0, DWORD dwSize = 0) noexcept
407  {
408  assert( m_hChildStdoutRd == NULL && m_hChildStdinWr == NULL );
409 
410  HANDLE hChildStdoutRd = NULL;
411  HANDLE hChildStdoutWr = NULL;
412  HANDLE hChildStdinRd = NULL;
413  HANDLE hChildStdinWr = NULL;
414 
415  BOOL bRes = TRUE;
416  (void)bRes;
417 
418  // Set the bInheritHandle flag so pipe handles are inherited.
419  SECURITY_ATTRIBUTES saAttr;
420  saAttr.nLength = sizeof(SECURITY_ATTRIBUTES);
421  saAttr.bInheritHandle = TRUE;
422  saAttr.lpSecurityDescriptor = NULL;
423 
424  // Create a pipe for the child process's STDOUT.
425  if ( !::CreatePipe(&hChildStdoutRd, &hChildStdoutWr, &saAttr, dwSize) )
426  return FALSE; //::GetLastError()
427 
428  // Ensure the read handle to the pipe for STDOUT is not inherited.
429  if ( !::SetHandleInformation(hChildStdoutRd, HANDLE_FLAG_INHERIT, 0) ) {
430  //FALSE, ::GetLastError()
431  bRes = ::CloseHandle(hChildStdoutRd);
432  assert( bRes );
433  bRes = ::CloseHandle(hChildStdoutWr);
434  assert( bRes );
435  return FALSE;
436  }
437 
438  // Create a pipe for the child process's STDIN.
439  if ( !::CreatePipe(&hChildStdinRd, &hChildStdinWr, &saAttr, dwSize) ) {
440  //FALSE, ::GetLastError()
441  bRes = ::CloseHandle(hChildStdoutRd);
442  assert( bRes );
443  bRes = ::CloseHandle(hChildStdoutWr);
444  assert( bRes );
445  return FALSE;
446  }
447 
448  // Ensure the write handle to the pipe for STDIN is not inherited.
449  if ( !::SetHandleInformation(hChildStdinWr, HANDLE_FLAG_INHERIT, 0) ) {
450  //FALSE, ::GetLastError()
451  bRes = ::CloseHandle(hChildStdoutRd);
452  assert( bRes );

454  assert( bRes );

461 
462  // Create the child process.
463  m_siStartupInfo.dwFlags |= STARTF_USESTDHANDLES;
464  m_siStartupInfo.hStdInput = hChildStdinRd;
465  m_siStartupInfo.hStdOutput = hChildStdoutWr;
466  m_siStartupInfo.hStdError = hChildStdoutWr;
467  if ( !baseClass::Create(szCmdLine, TRUE, szWorkPath, dwFlags) ) {
468  bRes = ::CloseHandle(hChildStdoutRd);
469  assert( bRes );
470  bRes = ::CloseHandle(hChildStdoutWr);
