755 inline
756 bool host_to_sockaddr(const WCHAR* host, UINT port,
757  SOCKADDR_STORAGE& ss, int& len) noexcept
758 {
759  WCHAR* szHost = ::_wcsdup(host);
760  if ( szHost == NULL )
761  return false;
762  if ( !_w_socket_helper::FillSockAddr(ss, len, AF_INET, port, szHost) ) {
763  ::free(szHost);
764  _w_socket_addr sa;
765  WCHAR szPort[64] = { 0 };
766  ::_snwprintf_s(szPort, sizeof(szPort) / sizeof(WCHAR), _TRUNCATE, L"%u", port);
767  if ( !sa.Fetch(host, szPort) )
768  return false;
769  const ADDRINFOEXW* pInfo = sa.GetAddrInfo();
770  while ( pInfo != NULL ) {
771  if ( pInfo->ai_family == AF_INET ) {
772  ::memcpy(&ss, pInfo->ai_addr, pInfo->ai_addrlen);
773  len = (int)pInfo->ai_addrlen;
774  return true;
775  }
776  pInfo = pInfo->ai_next;
777  }
778  return false;
779  }
780  ::free(szHost);
781  return true;
782 }
