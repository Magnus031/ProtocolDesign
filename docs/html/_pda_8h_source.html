263  //process event
264  void process_event(int iCurrentState, uint uEventNo) throw()
265  {
266  assert( !_Is_Pda_Table_Null(m_table) );
267  assert( iCurrentState >= PDA_STATE_START );
268  //look table
269  const PDA_TRANSITION_ITEM* pItem = m_table.pState[iCurrentState].pTransition;
270  if( pItem == NULL ) {
271  m_iCurrentActState = m_table.pState[iCurrentState].iDefaultState;
272  return ;
273  }
274  while( pItem->uEventFirstNo != PDA_LAST_EVENT_NO ) {
275  if( uEventNo >= pItem->uEventFirstNo && uEventNo <= pItem->uEventLastNo ) {
276  m_iCurrentActState = pItem->iNextState;
277  return ;
278  }
279  pItem ++;
280  } //end while
281  m_iCurrentActState = m_table.pState[iCurrentState].iDefaultState;
282  }

289  //revert
290  bool revert_with_current_event(bool bFromTop, uintptr& uRemovedNum) throw()
291  {
292  assert( m_uCurrentEvent != PDA_NO_EVENT );
293  uRemovedNum = 0;
294  //from top to bottom
295  auto iter(m_stack.GetTail());
296  if( !bFromTop )
297  iter.MovePrev();
298  while( iter != m_stack.GetEnd() ) {
299  //try
300  process_event(iter.get_Value().m_iState, m_uCurrentEvent);
301  if( !IsError() ) {
302  //remove
303  iter.MoveNext();
304  while( iter != m_stack.GetEnd() ) {
305  auto iter1(iter);
306  iter.MoveNext();
307  m_stack.RemoveAt(iter1);
308  uRemovedNum ++;
309  }
310  return true;
311  }
312  iter.MovePrev();
313  } //end while
314  //clear
315  m_uCurrentEvent = PDA_NO_EVENT;
316  return false;
317  }
