38  _os_com_ptr<IWbemClassObject> sp;
39  bool bContinue = hwmi.EnumFirst(L"MSFT_VSInstance", sp);
40  for ( ; bContinue; bContinue = hwmi.EnumNext(sp) ) {
41  _os_com_variant varCaption, varInstallLocation;
42  hr = sp->Get(L"Caption", 0, &varCaption, NULL, NULL);
43  if ( FAILED(hr) )
44  continue;
45  if ( varCaption.vt != VT_BSTR )
46  continue;
47  hr = sp->Get(L"InstallLocation", 0, &varInstallLocation, NULL, NULL);
48  if ( FAILED(hr) )
49  continue;
50  if ( varInstallLocation.vt != VT_BSTR )
51  continue;
52 
53  _os_com_bstr bstrCaption(varCaption.bstrVal);
54  varCaption.vt = VT_EMPTY;
55  int iYear = 0;
56  UINT uLength = bstrCaption.GetLength();
57  if ( uLength < 4 )
58  continue;
59  for ( UINT i = uLength - 4; i < uLength; i ++ ) {
60  if ( !char_is_digit(bstrCaption[i]) ) {
61  iYear = -1;
62  break;
63  }
64  iYear = iYear * 10 + (bstrCaption[i] - L'0');
65  }
66  if ( iYear == -1 )
67  continue;
68  if ( iYear > m_iMaxYear ) {
69  m_iMaxYear = iYear;
70  m_varInstallLocation = varInstallLocation; // may throw
71  }
72  }
73 
74  if ( m_iMaxYear == 0 )
75  return false;
76  _os_com_bstr bstrInstallLocation(m_varInstallLocation.bstrVal);
77  m_cstrDir = const_string_s(bstrInstallLocation, bstrInstallLocation.GetLength());
78  bstrInstallLocation.Detach();
79  return true;
80  }
