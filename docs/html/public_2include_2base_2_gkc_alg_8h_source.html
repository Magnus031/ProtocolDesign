67  //0 -- sunday, ...
68  static int CalcWeekDay(int iYear, int iMonth, int iDay) noexcept
69  {
70  int ret = CompareDeletedDate(iYear, iMonth, iDay);
71  if ( iMonth == 1 || iMonth == 2 ) {
72  -- iYear;
73  iMonth += 12;
74  }
75  int c = iYear / 100;
76  int y = iYear - 100 * c;
77  int w = y + y / 4 + (13 * (iMonth + 1) / 5) + iDay - 1;
78  if ( ret < 0 )
79  w = w + 5 - c;
80  else
81  w = w + c / 4 - 2 * c;
82  return (w % 7 + 7) % 7;
83  }

85  //0-365
86  static int CalcDayOrdinal(int iYear, int iMonth, int iDay, bool bLeap) noexcept
87  {
88  if ( iMonth < 2 )
89  return iDay - 1;
90  int count = 0;
91  for ( int i = 1; i < iMonth; ++ i )
92  count += CalcMonthDays(i, bLeap);
93  count += (iDay - 1);
94  int ret = CompareDeletedDate(iYear, iMonth, iDay);
95  if ( iYear == 1582 && ret >= 0 )
96  count -= ret; //deleted days
97  return count;
98  }

122  int iYearE, int iMonthE, int iDayE) noexcept
123  {

148 class CRC32Util
149 {
150 protected:
155  enum {
156  CRC_MASK = 0xFFFFFFFFUL,
157  CRC_32_POLYNORMAL = 0xEDB88320UL,
158  CRC_FIRST = -1L

162  CRC32Util() throw()

167 protected:
172  void build_table() throw()
173  {
174  //Initialize
175  uint i, j;
176  uint value;
177  for(i = 0; i < 256; i ++) {
178  value = i;
179  for(j = 8; j > 0; j --) {
180  if(NumberHelper::IsOdd<uint>(value))
181  value = (value >> 1) ^ CRC_32_POLYNORMAL;
182  else
183  value >>= 1;
184  }
185  m_table[i] = value;
186  } //end for
187  }
188 
189 protected:
198  uint update_block_crc32(const byte* lpBuf, uintptr uCount, uint uCRC) const throw()
199  {
200  const byte* p = lpBuf;
201  uint tp1, tp2;
202 
203  while( uCount-- != 0 ) {
204  tp1 = (uCRC >> 8) & 0x00FFFFFFUL;
205  tp2 = m_table[ (uCRC ^ ((uint)(*p))) & 0x000000FFUL ];
206  uCRC = tp1 ^ tp2;
207  ++ p;
208  } //end while
209 
210  return uCRC;
211  }
212 
213 protected:
218  uint m_table[256];
219 
220 private:
221  friend class CRC32Calc;
222 };
