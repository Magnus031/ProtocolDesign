214  template <class Tstring>
215  static void AppendCurrentPathPrefix(Tstring& str)
216  {
217  uintptr uLength = str.GetLength();
218  if( uLength == 0 )
219  return ;
220  for( auto iter(str.GetBegin()); iter != str.GetEnd(); iter.MoveNext() ) {
221  const typename Tstring::EType& ch = iter.get_Value();
222  if( check_path_separator(ch) || check_drive_separator(ch) )
223  return ;
224  }
225  const typename Tstring::EType* sz;
226  get_current_path_prefix(sz, uLength);
227  ConstStringT<typename Tstring::EType> c_strPrefix(sz, uLength);
228  StringOpHelper::Insert(0, c_strPrefix, str); //may throw
229  }
