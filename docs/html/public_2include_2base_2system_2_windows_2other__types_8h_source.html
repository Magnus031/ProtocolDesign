31 inline void get_tick_count(time_value& tv) throw()
32 {
33  LARGE_INTEGER liCount;
34  BOOL bRet = ::QueryPerformanceCounter(&liCount);
35  (void)bRet;
36  assert( bRet ); //FALSE, ::GetLastError()
37  //s
38  int64 iResult = liCount.QuadPart / _os_g_time_liFreq.QuadPart;
39  liCount.QuadPart -= (iResult * _os_g_time_liFreq.QuadPart);
40  tv.set_Seconds(iResult);
41 /*
42 Because liCount.QuadPart(rest) < 1,000,000,000 (ns),
43 so liCount.QuadPart * (1000 * 1000 * 1000) < limits_base<int64>::Max
44 */
45  //ns
46  tv.set_Nanoseconds(liCount.QuadPart * (1000 * 1000 * 1000) / _os_g_time_liFreq.QuadPart);
47 }

69 inline void _os_timevalue_to_filetime(const time_value& tv, FILETIME& ft) throw()
70 {
71  int64 v = (tv.get_Seconds() + _TIME_DELTA_1601_1970) * _TIME_SECOND_100_NS

75 inline void _os_filetime_to_timevalue(const FILETIME& ft, time_value& tv) throw()
76 {
77  int64 v = _os_filetime_to_100ns(ft);
78  int64 s = v / _TIME_SECOND_100_NS;
79  tv.Initialize(s - _TIME_DELTA_1601_1970, (v - s * _TIME_SECOND_100_NS) * 100);
80 }

109 inline bool make_local_time(int iYear, int iMonth, int iDay,
110  int iHour, int iMinute, int iSecond, int64 iNanoseconds,
111  time_value& tv) throw()
112 {
113  assert( iSecond >= 0 && iSecond <= 59 );
114  assert( iNanoseconds >= 0 && iNanoseconds < TIME_SECOND_FULL_NS );
115 
116  struct tm tms;
117  _fill_tm_from_detail(iYear, iMonth, iDay,

120  __time64_t tmu = ::_mktime64(&tms);
