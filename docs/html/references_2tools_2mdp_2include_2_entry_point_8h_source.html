42 int _Cmd_ProcessProjectFile(const ConstStringS& cstrSrc, const ConstStringS& cstrDest, int type, bool bLatest)
43 {
44  StringS strSrc(StringHelper::MakeEmptyString<CharS>(MemoryHelper::GetCrtMemoryManager()));
45  StringUtilHelper::MakeString(cstrSrc, strSrc);
46  StringS strSrcDir(StringHelper::MakeEmptyString<CharS>(MemoryHelper::GetCrtMemoryManager()));
47  StringUtilHelper::MakeString(cstrSrc, strSrcDir);
48  StringS strDest(StringHelper::MakeEmptyString<CharS>(MemoryHelper::GetCrtMemoryManager()));
49  {
50  uintptr uPos;
51  //source
52  if( !_CheckFileExtension(StringUtilHelper::To_ConstString(strSrc), DECLARE_TEMP_CONST_STRING(ConstStringS, _S(".mdp")), uPos) ) {
53  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Command error: The project file name must have the extension \".mdp\"!")));
54  return 1;
55  }
56  //source directory
57  uPos = FsPathHelper::FindFilePartStart(StringUtilHelper::To_ConstString(strSrcDir));
58  FsPathHelper::ToPathPart(uPos, strSrcDir); //delete file part: xxx.mdp
59  if( strSrcDir.IsEmpty() )
60  StringUtilHelper::MakeString(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("./")), strSrcDir);
61  //destination
62  if( !cstrDest.IsNull() ) {
63  StringUtilHelper::MakeString(cstrDest, strDest);
64  }
65  else {
66  if( !FileManagementHelper::GetCurrentDirectory(strDest) ) {
67  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Command error: The current directory cannot be obtained!")));
68  return 1;
69  }
70  FsPathHelper::AppendSeparator(strDest);
71  StringUtilHelper::Append(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("mdb-output")), strDest);
72  }
73  } //end block
74  FsPathHelper::AppendSeparator(strDest);
75  if( type == MDP_TYPE_CHM )
76  StringUtilHelper::Append(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("chm")), strDest);
77  else if( type == MDP_TYPE_EPUB )
78  StringUtilHelper::Append(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("epub")), strDest);
79  else
80  assert( false );
81 
82  //directory/file names
83  FsPathHelper::ConvertPathStringToPlatform(strSrc);
84  FsPathHelper::ConvertPathStringToPlatform(strDest);
85 
86  //create directory
87  if( !FileManagementHelper::ForceDirectory(strDest) ) {
88  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Command error: The destination directory cannot be created!")));
89  return 1;
90  }
91 
92  _PrintVersion();
93 
94  //process
95  return _Process_Project_File(strSrc, strSrcDir, strDest, type, bLatest) ? 0 : 2;
96 }

104  {

112  }

115  uintptr uStartIndex = 2;
116  //-m

118  if( uArgCount > 4 ) {
119  _PrintVersion();
120  _PrintHelp();
121  return 1;
122  }
123  type = MDP_TYPE_CHM;
124  }
125  //-e
126  else if( ConstStringCompareTrait<ConstStringS>::IsEQ(args[1], DECLARE_TEMP_CONST_STRING(ConstStringS, _S("-e"))) ) {
127  //-l
128  if( ConstStringCompareTrait<ConstStringS>::IsEQ(args[2], DECLARE_TEMP_CONST_STRING(ConstStringS, _S("-l"))) )
129  uStartIndex ++;
130  if( (uStartIndex == 2 && uArgCount > 4) || (uStartIndex == 3 && (uArgCount == 3 || uArgCount > 5)) ) {
131  _PrintVersion();
132  _PrintHelp();
133  return 1;
134  }
135  type = MDP_TYPE_EPUB;
136  }
137  else {
138  ConsoleHelper::WriteLine(DECLARE_TEMP_CONST_STRING(ConstStringS, _S("Command error: Invalid parameters!")));
139  return 1;
140  }
141 
142  return _Cmd_ProcessProjectFile(args[uStartIndex],
143  (uStartIndex + 1 >= uArgCount) ? ConstStringS() : args[uStartIndex + 1],
144  type, uStartIndex == 3);
145  }
