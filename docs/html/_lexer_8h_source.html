59  void Finish()
60  {
61  uint uCount = m_symbol_pool.GetTotalCount();
62  //may be an empty table
63  if( uCount == 0 )
64  return ;
65  //find minimum & maximum ID
66  uint uMinID = Limits<uint>::Max;
67  uint uMaxID = 0;
68  {
69  auto iter(m_symbol_pool.GetZeroLevelHead());
70  for( ; !iter.IsNull(); iter.MoveLevelNext() ) {
71  uint uID = iter.GetData<BeType<uint>>().get_Value();
72  if( uID < uMinID )
73  uMinID = uID;
74  if( uID > uMaxID )
75  uMaxID = uID;
76  }
77  } //end block
78  assert( uMinID != 0 && uMaxID != 0 && uMinID <= uMaxID );
79  uCount = uMaxID - uMinID;
80  uCount = SafeOperators::AddThrow(uCount, (uint)1); //may throw
81  //allocate
82  uint uBytes = SafeOperators::MultiplyThrow(uCount, (uint)sizeof(uint)); //may throw
83  uint uAddr = m_str_allocator.Allocate(uBytes);
84  if( uAddr == 0 )
85  throw OutOfMemoryException();
86  m_uIDFirstAddr = uAddr;
87  m_uMinID = uMinID;
88  m_uMaxID = uMaxID;
89  //fill
90  auto iter(m_symbol_pool.GetZeroLevelHead());
91  for( ; !iter.IsNull(); iter.MoveLevelNext() ) {
92  uint uID = iter.GetData<BeType<uint>>().get_Value();
93  uAddr = m_uIDFirstAddr + sizeof(uint) * (uID - uMinID);
94  _RefAllocatorHelper::ToObject<BeType<uint>>(RefPtrHelper::TypeCast<ArrayPoolAllocator, IMemoryAllocatorRef32>(RefPtr<ArrayPoolAllocator>(m_str_allocator)), uAddr).set_Value(iter.GetKeyAddr());
95  } //end for

206  // called repeatedly.
207  // return : SystemCallResults::OK, the token id may be CPL_TK_ERROR.
208  // SystemCallResults::S_False, it reaches the end of stream.
209  // otherwise, this call is failed.
210  CallResult Parse()
211  {
212  CallResult cr;
213 
214  bool bErrorOrEnd;
215  //can restart (last token is error or EOE)
216  if( !m_fsa.CanRestart(bErrorOrEnd) ) {
217  cr.SetResult(bErrorOrEnd ? SystemCallResults::Fail : SystemCallResults::S_False);
218  return cr;
219  }
220 
221  uint uEvent;
222  CharF ch;
223  //read a character
224  cr = m_stream.Deref().GetChar(ch);
225  if( cr.IsFailed() )
226  return cr;
227  if( cr.GetResult() == SystemCallResults::S_EOF ) {
228  cr.SetResult(SystemCallResults::S_False);
229  return cr;
230  }
231 
232  //start state
233  m_fsa.SetStartState();
234  m_token_info.Reset(); //may throw
235  //first character
236  uEvent = (uint)ch;
237  m_token_info.Append(ch); //may throw
238 
239  //loop
240  do {
241  //state
242  m_fsa.ProcessState(uEvent);
243  //stopped
244  if( m_fsa.IsStopped() ) {
245  //match
246  uint uBackNum;
247  int iMatch = m_fsa.GetMatch(uBackNum);
248  assert( iMatch >= 0 );
249  if( iMatch == 0 ) {
250  //error
251  m_token_info.set_ID(CPL_TK_ERROR);
252  break;
253  }
254  //iMatch is the same as token id (from CPL_TK_FIRST)
255  m_token_info.set_ID(iMatch);
256  //back
257  if( uBackNum > 0 ) {
258  cr = m_stream.Deref().UngetChar((int64)uBackNum);
259  assert( cr.IsOK() );
260  m_token_info.BackChar(uBackNum); //may throw
261  }
262  uint uID = m_token_info.get_ID();
263  //action
264  ShareCom<_ILexerAction> action(find_action(uID));
265  if( !action.IsBlockNull() ) {
266  ConstStringA strToken(m_token_table.Deref().get_Token(uID));
267  bool bTokenChanged = false;
268  cr = action.Deref().DoAction(m_stream, m_token_info, strToken, bTokenChanged);
269  if( cr.IsFailed() ) {
270  //process CPL_TK_ERROR
271  if( m_token_info.get_ID() == CPL_TK_ERROR )
272  cr.SetResult(SystemCallResults::OK);
273  break;
274  }
275  if( bTokenChanged ) {
276  uID = m_token_table.Deref().get_ID(strToken);
277  assert( uID >= CPL_TK_FIRST );
278  m_token_info.set_ID(uID);
279  }
280  } //end if
281  break;
282  } //end if
283  //get next char
284  cr = m_stream.Deref().GetChar(ch);
285  if( cr.IsFailed() )
286  break;
287  if( cr.GetResult() == SystemCallResults::S_EOF ) {
288  uEvent = FSA_END_OF_EVENT;
289  cr.SetResult(SystemCallResults::OK);
290  }
291  else {
292  uEvent = (uint)ch;
293  m_token_info.Append(ch); //may throw
294  }
295  } while( true );
296 
297  return cr;
298  }
